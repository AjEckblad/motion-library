Honey pour

Honey pour is a WebGL effect: liquid-rope coiling: a tapering thread lays rope loops that fillet into a thin-film pool. One plain JavaScript file with no dependencies; add it with npx motion-library add honey-coil.

Open it live, tune it and copy the code

What is Honey pour?

Honey pours from a still spoon into a porcelain bowl and coils into a stacked column that sinks into the pool, now and then toppling over. Drag to move the spoon and stitch a trail of loops.

How does Honey pour work?

Flow sets the thread thickness, loops the coils per second, spread how fast the mound slumps; drag the pointer to stitch trails of loops.

How do I add Honey pour to my website?

  1. In your project folder, run npx motion-library add honey-coil. It writes animations/honey-coil.js. Needs Node 18 or newer.
  2. Import it and call it with a container element that has a size: import fx from './animations/honey-coil.js'; const stop = fx(document.querySelector('#hero'));
  3. Call stop() to clean up, for example when a component unmounts.

Every setting below is also a flag, so you can install it already tuned: npx motion-library add honey-coil --flow 1.5.

// Honey pour: a thread of honey falls into a white porcelain bowl of honey on dark oiled wood, lit like a studio
// product shot. The thread falls straight and plumb under the spoon, thinning as it speeds up (radius ~ v^-1/2), and only
// its last few millimetres wind out from the axis into a helix that lays the rope down tangent to the loop it is making
// (liquid-rope coiling, as in Ribe's and Maleki's experiments): a steady pour builds a neat stacked coil of a few loops
// that rises as a column while its base sinks into the pool under the column's weight; now and then the column topples
// and starts again beside its heap, or the rope flattens into folding back and forth and may come out of it turning the
// other way. Move the pointer and the pour follows it: dragged, the coil trails the spoon by up to a few coil radii and
// is laid out as a trail of loops, like a sewing machine stitching in honey.
// The trick: the coil is kinematic but its pile is physical. Each fresh loop is a chain of capsules whose points rest
// either on the pool, sinking into it faster the more rope stands on them (and faster still on half-melted rope, too
// soft to hold anything up), or on the loop beneath them, squashed slightly into it. Once the bottom loop's centre has
// sunk just below the surface it is melted into a GPU height field as exactly the part that still shows, joined to the
// surface by the same fillet the live rope had, and fades into a mound on the CPU: thin-film (lubrication) spreading,
// solved implicitly, plus a relaxation that stands for the deep honey carrying the excess away under the surface, so
// the mound stays a skirt round the pour and the rest of the pool lies level (volume in equals volume out; the mound
// never grows without bound). Each pixel raymarches the thread, the loops and the surface as one liquid (ropes joined
// to the pool by a smooth union) and the bowl analytically: two half-ellipsoids and a rolled lip (a sphere-traced
// torus); honey wets the glaze, so the surface climbs it in a meniscus at the shore. The coil's capsules sit in groups
// of four behind bounding spheres, and each ray first marks the groups (and thread segments) it passes near.
// Honey's colour is Beer-Lambert absorption along the light's whole path: down through the honey to the porcelain and
// back up, across a rope and on down into the pool or out into the air (refracted, so each loop shows a bright band
// over a darker one), pale gold where it is thin, deep amber where it is deep, with a little light scattered inside.
// The studio is dark: two narrow strip softboxes behind (crisp bars in the pool that bend round the skirt of the pile),
// a square box overhead (the sheen along the loops), a dim one over the camera (catchlights, the sheen on the glaze),
// a warm light on the table behind, and two cards beside the bowl for the crisp lines along the thread.
const PARAMS = { flow: 1, loops: 1.5, spread: 1 };

export default function (stage) {
  const canvas = document.createElement('canvas');
  canvas.style.cssText = 'position:absolute;inset:0;width:100%;height:100%;display:block;touch-action:pan-y';
  stage.append(canvas);
  const gl = canvas.getContext('webgl2', { antialias: false, depth: false, alpha: false, powerPreference: 'high-performance' });
  if (!gl) { stage.textContent = 'This needs WebGL2.'; return () => canvas.remove(); }
  if (!gl.getExtension('EXT_color_buffer_float')) { stage.textContent = 'This needs WebGL2 float render targets.'; gl.getExtension('WEBGL_lose_context')?.loseContext(); return () => canvas.remove(); }
  const linF = !!gl.getExtension('OES_texture_float_linear');
  const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;

  // ---------------------------------------------------------------- the physical set-up, in millimetres
  const H0 = 4.4;                                   // depth of the pool at the middle of the bowl
  const FLOW = Math.max(0.5, Math.min(1.5, PARAMS.flow));
  const AL = 1.0 * Math.sqrt(FLOW), A1 = 0.55 * Math.sqrt(FLOW);   // laid rope radius, thread radius at its neck
  const FREQ = Math.max(0.8, Math.min(2.5, PARAMS.loops)), RC0 = 3.1;
  const TAU = 2.0;                                  // a melted loop lingers as a soft ridge for a few seconds
  const TRB = 1.2, KB = 0.035 * Math.max(0.6, Math.min(2.5, PARAMS.spread));  // rho g / (3 mu), for a cold, thick honey
  const NB = 160, LB = 80, DXB = LB / NB;           // the mound: 80 mm square on the CPU
  const NR = 384, LR = 32;                          // the fresh rope: 32 mm square on the GPU

  // ---------------------------------------------------------------- GLSL
  const VQ = `#version 300 es
  in vec2 p; void main() { gl_Position = vec4(p, 0, 1); }`;
  const HEAD = `#version 300 es
  precision highp float; precision highp int; precision highp sampler2D;
  `;
  // fresh rope: blur a little and fade (its volume is handed to the mound on the CPU), then lay down the new rope
  const FS_BLUR = HEAD + `
  uniform sampler2D S; uniform vec2 DIR; uniform float K, BL; out vec4 o;
  void main() {
    ivec2 p = ivec2(gl_FragCoord.xy), n = textureSize(S, 0) - 1; ivec2 d = ivec2(DIR);
    float c = texelFetch(S, p, 0).r * .375 + (texelFetch(S, clamp(p + d, ivec2(0), n), 0).r + texelFetch(S, clamp(p - d, ivec2(0), n), 0).r) * .25
      + (texelFetch(S, clamp(p + 2 * d, ivec2(0), n), 0).r + texelFetch(S, clamp(p - 2 * d, ivec2(0), n), 0).r) * .0625;
    o = vec4(mix(texelFetch(S, p, 0).r, c, BL) * K, 0, 0, 1);
  }`;
  const FS_DEP = HEAD + `
  uniform sampler2D S, TB; uniform vec4 SEG[8]; uniform vec2 SY[8]; uniform int NS; uniform float LR, LB; out vec4 o;
  void main() {
    ivec2 p = ivec2(gl_FragCoord.xy); float n = float(textureSize(S, 0).x);
    vec2 x = (vec2(p) + .5) / n * LR - LR * .5;
    float h = texelFetch(S, p, 0).r, b = texture(TB, x / LB + .5).r;
    for (int i = 0; i < 8; i++) {
      if (i >= NS) break;
      vec2 a = SEG[i].xy, ab = SEG[i].zw - a; float l = dot(ab, ab);
      float t = l > 1e-9 ? clamp(dot(x - a, ab) / l, 0., 1.) : 0.;
      float d = length(x - a - ab * t), r = SY[i].x;
      if (d > r + .8) continue;
      // the upper half of the rope as it stands, joined to what is there by the same fillet the fresh rope had
      float top = SY[i].y - b + (d < r ? sqrt(r * r - d * d) : -(d - r) * 3.), k = .45, w = clamp(.5 + .5 * (top - h) / k, 0., 1.);
      h = mix(h, top, w) + k * w * (1. - w) * .5;
    }
    o = vec4(h, 0, 0, 1);
  }`;

  const FS_MAIN = HEAD + `
  uniform sampler2D TB, TR; uniform vec4 BZ[14]; uniform vec4 BS; uniform vec3 NZ; uniform float YA; uniform vec4 LV[128]; uniform vec4 LG[32]; uniform vec4 LS; uniform int NLV;
  uniform float A1, HMAX, H0, LB, LR, EXPO, HL, RS; uniform vec2 CC; uniform float CA; uniform vec4 STK;
  uniform vec3 CP, CF, CR, CU; uniform float FOC; uniform vec2 R; out vec4 o;
  const vec3 SIG = vec3(.04, .155, .5);                   // honey's absorption per mm: blue goes first, then green
  const float ETA = 1.49, KF = .6;                       // KF: how wide the honey fillets a rope into the pool
  const vec3 LD = vec3(0., .94, -.342);                  // the main light: overhead, a little behind
  // the dish: white porcelain, inside and outside half-ellipsoids of revolution, standing on a dark table
  const float DA = 30., DB = 10., DC = 10., OA = 32.2, OB = 13., YRIM = 7.5, YT = -3., RIN = 29.05, ROUT = 31.6;
  const float RMM = (RIN + ROUT) * .5, RRM = (ROUT - RIN) * .5;
  const vec2 SPC = vec2(0., -7.);
  float yFl(vec2 x) { return DC - DB * sqrt(max(1. - dot(x, x) / (DA * DA), 0.)); }
  float lightE(vec2 x) { vec2 s = x - SPC; float q = dot(s, s); return exp(-q / 800.) * .9 + .1 * exp(-q / 8000.); }
  float bsp(sampler2D T, vec2 uv) {                     // cubic B-spline from four bilinear taps: smooth slopes
    vec2 n = vec2(textureSize(T, 0)), u = uv * n - .5, i = floor(u), f = u - i;
    vec2 w0 = (1. - f) * (1. - f) * (1. - f) / 6., w1 = (4. - 6. * f * f + 3. * f * f * f) / 6., w2 = (1. + 3. * f + 3. * f * f - 3. * f * f * f) / 6., w3 = f * f * f / 6.;
    vec2 g0 = w0 + w1, g1 = w2 + w3, h0 = (i - .5 + w1 / g0) / n, h1 = (i + 1.5 + w3 / g1) / n;
    return g0.y * (g0.x * texture(T, vec2(h0.x, h0.y)).r + g1.x * texture(T, vec2(h1.x, h0.y)).r)
         + g1.y * (g0.x * texture(T, vec2(h0.x, h1.y)).r + g1.x * texture(T, vec2(h1.x, h1.y)).r);
  }
  float hB(vec2 x) { return bsp(TB, x / LB + .5); }
  float hBf(vec2 x) { return texture(TB, x / LB + .5).r; }
  float hR(vec2 x) { vec2 u = x / LR + .5; return (u.x < 0. || u.y < 0. || u.x > 1. || u.y > 1.) ? 0. : texture(TR, u).r; }
  float hRs(vec2 x) { vec2 u = x / LR + .5; return (u.x < 0. || u.y < 0. || u.x > 1. || u.y > 1.) ? 0. : bsp(TR, u); }
  // honey wets the glaze: at the shore the surface climbs the porcelain in a meniscus, then meets it at a low angle
  float men(vec2 x) { float q = dot(x, x); if (q < (RS - 3.) * (RS - 3.)) return 0.; float e = sqrt(q) - RS; return e < 0. ? .2 * exp(e / .45) : .2 + .3 * e; }
  float H(vec2 x) { vec2 q = x - CC; return hB(x) + hRs(x) + CA * .55 * exp(-dot(q, q) / (CA * CA * 2.2)) + men(x); }
  float Hm(vec2 x) { vec2 q = x - CC; return hBf(x) + hR(x) + CA * .55 * exp(-dot(q, q) / (CA * CA * 2.2)) + men(x); }
  vec3 Hn(vec2 x, float e) { float h = H(x); return normalize(vec3(h - H(x + vec2(e, 0)), e, h - H(x + vec2(0, e)))); }
  vec3 Hnf(vec2 x, float e) { float a = Hm(x + vec2(e, 0)), b = Hm(x - vec2(e, 0)), c = Hm(x + vec2(0, e)), d = Hm(x - vec2(0, e)); return normalize(vec3(b - a, 2. * e, d - c)); }
  // ---- the studio: dark and warm; two strip softboxes behind, a square one overhead, a dim one over the camera, a card left
  float box(vec3 r, vec4 b, float s) {
    float el = asin(clamp(r.y, -1., 1.)), az = atan(r.x, -r.z);
    return smoothstep(b.x - s, b.x + s, el) * smoothstep(b.y + s, b.y - s, el) * smoothstep(b.z - s, b.z + s, az) * smoothstep(b.w + s, b.w - s, az);
  }
  vec3 studio(vec3 r, float sf) {
    float el = asin(clamp(r.y, -1., 1.)), az = atan(r.x, -r.z), s = .004 * sf;
    vec3 c = vec3(.008, .0058, .004) + vec3(.016, .011, .007) * smoothstep(-.1, .7, r.y);
    // two tall strip softboxes behind, left and right, crisp-edged, their diffusers a touch brighter in the middle
    float L = box(r, vec4(.3, .78, -.14, -.124), s), Q = box(r, vec4(.3, .78, .1, .111), s);
    c += vec3(3., 2.85, 2.65) * L * (.7 + .3 * sin(clamp((az + .14) / .016, 0., 1.) * 3.1416))
       + vec3(.9, .9, .95) * Q * (.7 + .3 * sin(clamp((az - .1) / .011, 0., 1.) * 3.1416));
    // a square softbox overhead, a little behind: the soft sheen along the tops of the loops
    c += vec3(1.5, 1.45, 1.4) * box(r, vec4(.98, 1.32, -.42, .42), .015 * sf);
    // a dimmer box up front, over the camera: catchlights on the near faces, the glaze's sheen on the far wall
    float fa = abs(az) - 3.1416 + .4;
    c += vec3(1.3, 1.28, 1.25) * smoothstep(-.004 * sf, .004 * sf, fa) * smoothstep(.62 - s, .62 + s, el) * smoothstep(1.02 + s, 1.02 - s, el);
    return c;
  }
  float card(vec3 o, vec3 r) {
    if (r.x > -.02) return -1.;
    float t = (-37. - o.x) / r.x; vec3 q = o + r * t;
    return (t > 0. && q.z > -45. && q.z < 30. && q.y > YT && q.y < 70.) ? t : -1.;
  }
  vec2 ell(vec3 ro, vec3 rd, float a, float b, float c) {
    float s = a / b; vec3 o = vec3(ro.x, (ro.y - c) * s, ro.z), d = vec3(rd.x, rd.y * s, rd.z);
    float A = dot(d, d), B = dot(o, d), C = dot(o, o) - a * a, h = B * B - A * C;
    if (h < 0.) return vec2(-1.);
    h = sqrt(h); return vec2(-B - h, -B + h) / A;
  }
  vec3 ellN(vec3 p, float a, float b) { return normalize(vec3(p.x / (a * a), (p.y - DC) / (b * b), p.z / (a * a))); }
  float traceDish(vec3 ro, vec3 rd, out vec3 n, out float m) {     // m: 1 porcelain, 2 table
    float t = 1e9; m = 0.; n = vec3(0, 1, 0);
    vec2 e = ell(ro, rd, OA, OB, DC);
    if (e.x > 0.) { vec3 p = ro + rd * e.x; if (p.y < YRIM) { t = e.x; n = ellN(p, OA, OB); m = 1.; } }
    if (abs(rd.y) > 1e-4) {                              // the rolled lip: a torus over the top of both walls
      float t0 = (YRIM + RRM - ro.y) / rd.y, t1 = (YRIM - RRM - ro.y) / rd.y;
      if (t0 > t1) { float q = t0; t0 = t1; t1 = q; }
      t0 = max(t0, 0.); t1 = min(t1, t);
      if (t0 < t1) {
        vec2 a = ro.xz + rd.xz * t0, ab = rd.xz * (t1 - t0); float u = clamp(-dot(a, ab) / max(dot(ab, ab), 1e-6), 0., 1.);
        if (max(length(a), length(a + ab)) > RIN - .3 && length(a + ab * u) < ROUT + .3) {
          float tt = t0;
          for (int i = 0; i < 28; i++) {
            vec3 p = ro + rd * tt; float r = length(p.xz); vec2 q = vec2(r - RMM, p.y - YRIM); float d = length(q) - RRM;
            if (d < .003) { t = tt; n = normalize(vec3(p.x / r * q.x, q.y, p.z / r * q.x)); m = 1.; break; }
            tt += d; if (tt > t1) break;
          }
        }
      }
    }
    if (rd.y < 0.) { float tt = (YT - ro.y) / rd.y; if (tt > 0. && tt < t) { t = tt; n = vec3(0, 1, 0); m = 2.; } }
    e = ell(ro, rd, DA, DB, DC);
    if (e.y > 0. && e.y < t) { vec3 p = ro + rd * e.y; if (p.y < YRIM) { t = e.y; n = -ellN(p, DA, DB); m = 1.; } }
    return m > 0. ? t : -1.;
  }
  vec3 shadeDish(vec3 p, vec3 rd, vec3 n, float m, float sf) {
    float E = lightE(p.xz), c = max(dot(n, -rd), 0.), F = .04 + .96 * pow(1. - c, 5.);
    if (m > 1.5) {                                        // dark oiled wood; the light dies away from the dish
      float ao = .3 + .7 * smoothstep(OA - 1., OA + 10., length(p.xz));
      vec2 g = p.xz * vec2(.9, .045);                     // a faint grain running across, so the gloss is not plastic
      float grain = .85 + .15 * sin(g.x + 2.3 * sin(g.y * 3.1 + p.x * .05)) * sin(g.y * 7. + p.x * .21);
      vec2 bq = p.xz - vec2(0., -78.); E += .55 * exp(-dot(bq, bq) / 1500.);   // a warm light on the table behind
      return vec3(.028, .016, .009) * grain * (1.3 * E + .03) * ao + F * .55 * studio(reflect(rd, n), 7.) * ao;
    }
    float dif = .72 * max(dot(n, LD), 0.) + .28 * (.55 + .45 * n.y);
    vec3 col = vec3(.86, .84, .8) * dif * E;
    col += vec3(.05, .03, .016) * E * (.5 + .5 * n.y);       // a little light scattered in the body of the porcelain
    float r = length(p.xz);                               // the honey's glow on the porcelain just above its edge
    if (r < RIN) col += vec3(.5, .24, .05) * E * exp(-max(p.y - HL, 0.) * .9) * smoothstep(RS - 2., RS + .5, r);
    return col * (1. - F) + F * studio(reflect(rd, n), sf) * 1.15;
  }
  vec3 poolFlat(vec3 q, vec3 r) {                         // the pool as seen in a reflection: level, at the shore's height
    float d = max(HL - yFl(q.xz), 0.), F = .04 + .96 * pow(1. - abs(r.y), 5.);
    vec3 t = refract(r, vec3(0, 1, 0), 1. / ETA); float L = d / max(-t.y, .3);
    return studio(reflect(r, vec3(0, 1, 0)), 2.) * F + (1. - F) * vec3(1., .95, .87) * 1.3 * lightE(q.xz) * exp(-SIG * (L + d * 1.1));
  }
  vec3 envScene(vec3 o, vec3 r, float sf) {
    vec3 n; float m, t = traceDish(o, r, n, m), tc = card(o, r);
    if (r.y < 0.) {
      float tp = (HL - o.y) / r.y; vec3 q = o + r * tp;
      if (tp > 0. && (t < 0. || tp < t) && (tc < 0. || tp < tc) && dot(q.xz, q.xz) < RS * RS) return poolFlat(q, r);
    }
    if (tc > 0. && (t < 0. || tc < t)) return vec3(1.15, 1.2, 1.3);
    if (t > 0.) return shadeDish(o + r * t, r, n, m, sf * 3.);
    return studio(r, sf);
  }
  // ---- the thread: a falling tube that thins as it speeds up (radius ~ v^-1/2), narrowest just above the coil
  float thR(float y) { return A1 * pow(1. + max(y - YA, 0.) / 9., .42); }
  float traceThread(vec3 ro, vec3 rd) {
    vec2 oc = ro.xz - NZ.xz, d = rd.xz; float a = dot(d, d), b = dot(oc, d);
    float r = A1 * 2.6, t = -1.;
    for (int k = 0; k < 3; k++) {
      float c = dot(oc, oc) - r * r, h = b * b - a * c; if (h < 0.) return -1.;
      t = (-b - sqrt(h)) / a; r = thR(ro.y + rd.y * t);
    }
    return (t > 0. && ro.y + rd.y * t > NZ.y) ? t : -1.;
  }
  float sdSeg(vec3 p, vec4 a, vec4 b) {
    vec3 pa = p - a.xyz, ba = b.xyz - a.xyz; float h = clamp(dot(pa, ba) / dot(ba, ba), 0., 1.);
    return length(pa - ba * h) - mix(a.w, b.w, h);
  }
  uint BM = 0xffffu;                                     // the bend's segments near the current ray (as LM below)
  float sdBend(vec3 p) {
    float d = length(p - BS.xyz) - BS.w; if (d > .3) return d;
    d = 1e5;
    for (int i = 0; i < 13; i++) if (((BM >> uint(i)) & 1u) != 0u) d = min(d, sdSeg(p, BZ[i], BZ[i + 1]));
    float top = length(p.xz - NZ.xz) - thR(p.y); d = min(d, max(top, NZ.y - p.y));
    return d;
  }

  float sdCap(vec3 p, vec3 a, vec3 b, float r) { vec3 pa = p - a, ba = b - a; float h = clamp(dot(pa, ba) / dot(ba, ba), 0., 1.); return length(pa - ba * h) - r; }
  // the fresh rope, as capsules in groups of four, each group bounded by a sphere. A ray first marks the groups its
  // path comes near (LM, a bitmask), then marching tests only those: the nearest first, then the ones nearer than that
  uint LM = 0xffffffffu;
  float sdLive(vec3 p) {
    float far = length(p - LS.xyz) - LS.w; if (far > .3) return far;
    int ng = (NLV + 2) / 4, bg = -1; float best = 1e5;
    for (int g = 0; g < 32; g++) {
      if (g >= ng) break;
      if (((LM >> uint(g)) & 1u) == 0u) continue;
      float e = length(p - LG[g].xyz) - LG[g].w; if (e < best) { best = e; bg = g; }
    }
    if (bg < 0 || best > .25) return best;
    float d = 1e5;
    for (int k = 0; k < 4; k++) { int i = bg * 4 + k; if (i >= NLV - 1) break; d = min(d, sdCap(p, LV[i].xyz, LV[i + 1].xyz, LV[i].w)); }
    for (int g = 0; g < 32; g++) {
      if (g >= ng) break;
      if (g == bg || ((LM >> uint(g)) & 1u) == 0u || length(p - LG[g].xyz) - LG[g].w > d) continue;
      for (int k = 0; k < 4; k++) { int i = g * 4 + k; if (i >= NLV - 1) break; d = min(d, sdCap(p, LV[i].xyz, LV[i + 1].xyz, LV[i].w)); }
    }
    return d;
  }
  bool rayNear(vec3 ro, vec3 rd, vec3 c, float r, float t0, float t1) {
    vec3 oc = ro - c; float b = dot(oc, rd), h = b * b - dot(oc, oc) + r * r;
    if (h < 0.) return false;
    h = sqrt(h); return -b + h > t0 && -b - h < t1;
  }
  void rayMasks(vec3 ro, vec3 rd, float pad, float t0, float t1) {
    LM = 0u; BM = 0u; int ng = (NLV + 2) / 4;
    for (int g = 0; g < 32; g++) { if (g >= ng) break; if (rayNear(ro, rd, LG[g].xyz, LG[g].w + pad, t0, t1)) LM |= 1u << uint(g); }
    for (int i = 0; i < 13; i++) {
      vec4 a = BZ[i], b = BZ[i + 1];
      if (rayNear(ro, rd, (a.xyz + b.xyz) * .5, length(b.xyz - a.xyz) * .5 + max(a.w, b.w) + pad, t0, t1)) BM |= 1u << uint(i);
    }
  }
  vec3 qb(vec3 a, vec3 b, vec3 c, float u) { return mix(mix(a, b, u), mix(b, c, u), u); }
  vec3 qd(vec3 a, vec3 b, vec3 c, float u) { return mix(b - a, c - b, u); }
  void liveFrame(vec3 p, out vec3 nrm, out vec3 ax) {
    float d = 1e5, tk = 0.; int k = 0;
    for (int i = 0; i < 127; i++) {
      if (i >= NLV - 1) break;
      vec3 pa = p - LV[i].xyz, ba = LV[i + 1].xyz - LV[i].xyz; float h = clamp(dot(pa, ba) / dot(ba, ba), 0., 1.), e = length(pa - ba * h);
      if (e < d) { d = e; k = i; tk = h; }
    }
    vec3 a = LV[k].xyz, b = LV[k + 1].xyz, c = mix(a, b, tk), tg = b - a;
    if (tk < .5 && k > 0) { vec3 m0 = (LV[k - 1].xyz + a) * .5, m1 = (a + b) * .5; c = qb(m0, a, m1, tk + .5); tg = qd(m0, a, m1, tk + .5); }
    else if (tk >= .5 && k < NLV - 2) { vec3 m0 = (a + b) * .5, m1 = (b + LV[k + 2].xyz) * .5; c = qb(m0, b, m1, tk - .5); tg = qd(m0, b, m1, tk - .5); }
    ax = normalize(tg); vec3 r = p - c; r -= ax * dot(r, ax); nrm = normalize(r + 1e-5);
  }
  void bendFrame(vec3 p, out vec3 nrm, out vec3 ax, out float rad) {
    float d = 1e5, tk = 0.; int k = 0;
    for (int i = 0; i < 13; i++) {
      vec3 pa = p - BZ[i].xyz, ba = BZ[i + 1].xyz - BZ[i].xyz; float h = clamp(dot(pa, ba) / dot(ba, ba), 0., 1.), e = length(pa - ba * h);
      if (e < d) { d = e; k = i; tk = h; }
    }
    vec3 a = BZ[k].xyz, b = BZ[k + 1].xyz, c = mix(a, b, tk), tg = b - a; rad = mix(BZ[k].w, BZ[k + 1].w, tk);
    if (tk < .5 && k > 0) { vec3 m0 = (BZ[k - 1].xyz + a) * .5, m1 = (a + b) * .5; c = qb(m0, a, m1, tk + .5); tg = qd(m0, a, m1, tk + .5); }
    else if (tk >= .5 && k < 12) { vec3 m0 = (a + b) * .5, m1 = (b + BZ[k + 2].xyz) * .5; c = qb(m0, b, m1, tk - .5); tg = qd(m0, b, m1, tk - .5); }
    ax = normalize(tg); vec3 r = p - c; r -= ax * dot(r, ax); nrm = normalize(r + 1e-5);

  }

  // ---- the honey surface: a height field, marched from the top of the tallest pile, inside the bowl only
  float traceH(vec3 ro, vec3 rd, int N) {

    for (const s of rsegs) {
      const ax = s.x1 - s.x0, az = s.z1 - s.z0, l = ax * ax + az * az, t = l > 1e-9 ? Math.max(0, Math.min(1, ((x - s.x0) * ax + (z - s.z0) * az) / l)) : 0;
      const dx = x - s.x0 - ax * t, dz = z - s.z0 - az * t, d2 = dx * dx + dz * dz;
      if (d2 < AL * AL) h = Math.max(h, (s.rel + Math.sqrt(AL * AL - d2)) * Math.exp(-s.age / TAU));
    }
    return h;
  };
  const poolAt = (x, z) => sampleB(x, z) + ropeAt(x, z) + AL;             // a rope's centre height lying on the pool
  let SPACING = 0.8, NEIGH = 4;               // rope point spacing (mm); chain neighbours closer than NEIGH are not "beneath"
  const loopAt = (x, z, jmax) => {            // centre height of a rope resting on the fresh loops under (x, z)
    let y = -1e9; const r2 = (2 * AL * SQ) ** 2;
    for (let j = 0; j < jmax; j++) { const q = live[j], d2 = (q.x - x) ** 2 + (q.z - z) ** 2; if (d2 < r2) y = Math.max(y, q.y + Math.sqrt(r2 - d2)); }
    return y;
  };
  const landY = (x, z) => Math.max(poolAt(x, z), loopAt(x, z, live.length - NEIGH));
  const DEP = new Float32Array(32), DEPY = new Float32Array(16); let nDep = 0;

  // the pour: a nozzle held still (or following the pointer), the coil centred under it, a contact point circling
  const noz = { x: 0, z: 0, vx: 0, vz: 0 }, coil = { phi: 0, x: RC0, z: 0, cx: 0, cz: 0, dir: 1, fold: false, ecc: 0 };
  const hash1 = n => { const x = Math.sin(n * 127.1 + 311.7) * 43758.5453; return x - Math.floor(x); };
  const sstep = (a, b, x) => { const t = Math.max(0, Math.min(1, (x - a) / (b - a))); return t * t * (3 - 2 * t); };
  const nz1 = (t, k) => { const i = Math.floor(t), f = t - i, u = f * f * (3 - 2 * f); return hash1(i + k * 57) * (1 - u) + hash1(i + 1 + k * 57) * u; };   // smooth 1D noise in 0..1
  let simT = 0, contactY = H0 + AL, pileH = 0;
  const topple = { t: -1, dx: 0, dz: 0 }; let nextTopple = 19;
  let px = 0.5, py = 0.5, lastMove = -1e9, held = false;
  function advance(dt, follow, target) {
    simT += dt;
    // left alone, the hand holding the spoon is steady: the nozzle barely drifts, so the rope builds a stacked coil
    const tx_ = follow ? target[0] : 1.6 * (nz1(simT * 0.03, 5) - 0.5), tz_ = follow ? target[1] : 1.6 * (nz1(simT * 0.027, 6) - 0.5);
    const sub = 4, h = dt / sub;
    SPACING = Math.max(0.8, 1.25 * AL); NEIGH = Math.ceil(3.4 * AL / SPACING);
    // the pile falls over now and then once it stands tall: its top slides off to one side
    if (topple.t < 0 && !follow && simT > nextTopple && pileH > 3.4 * AL) { const a = 6.283 * hash1(Math.floor(simT * 7) + 0.37); topple.t = 0; topple.dx = Math.cos(a); topple.dz = Math.sin(a); }
    const toff = topple.t < 0 ? 0 : sstep(0, 0.7, topple.t) * (1 - sstep(1.4, 4.5, topple.t));
    for (let s = 0; s < sub; s++) {
      noz.vx += (40 * (tx_ - noz.x) - 11 * noz.vx) * h; noz.vz += (40 * (tz_ - noz.z) - 11 * noz.vz) * h;
      noz.x += noz.vx * h; noz.z += noz.vz * h;
      // the coil is never quite steady: its radius and pace wander, it tightens as the pile rises toward the nozzle
      // (the fall is shorter), and now and then it flattens into folding, the rope swinging back and forth, and may come
      // out of it turning the other way, as real coiling honey does
      const pile = Math.min(1, Math.max(0, pileH / 6));
      const rc = RC0 * (0.86 + 0.24 * nz1(simT * 0.31, 1) + 0.03 * Math.sin(simT * 2.3)) * (1 - 0.2 * pile) * Math.sqrt(FLOW);
      coil.phi += coil.dir * 2 * Math.PI * FREQ * (0.9 + 0.2 * nz1(simT * 0.43, 2)) * (1 + 0.25 * pile) * h;
      const ecc = 0.88 * sstep(0.68, 0.86, nz1(simT * 0.11, 3)), psi = simT * 0.05 + 3 * nz1(simT * 0.05, 4); coil.ecc = ecc;
      if (ecc > 0.8 && !coil.fold) { coil.fold = true; if (hash1(Math.floor(simT * 0.11) + 7.3) > 0.5) coil.dir = -coil.dir; }
      else if (ecc < 0.4) coil.fold = false;
      // the coil centres under the nozzle; dragged, the rope trails behind it. A faint wander, the pile's own unevenness
      const vn = Math.hypot(noz.vx, noz.vz), lag = 0.32 * Math.min(1, 2.6 * RC0 / (0.32 * vn + 1e-6));   // the heel trails, up to a few coil radii
      let cx = noz.x - noz.vx * lag + 0.5 * rc * (nz1(simT * 0.13, 7) - 0.5) + topple.dx * toff * 1.1 * rc;
      let cz = noz.z - noz.vz * lag + 0.5 * rc * (nz1(simT * 0.12, 8) - 0.5) + topple.dz * toff * 1.1 * rc;
      cx = Math.max(-12, Math.min(12, cx)); cz = Math.max(-12, Math.min(12, cz)); coil.cx = cx; coil.cz = cz;
      const ux = rc * Math.cos(coil.phi), uz = rc * (1 - ecc) * Math.sin(coil.phi), cs = Math.cos(psi), sn = Math.sin(psi);
      coil.x = cx + ux * cs - uz * sn; coil.z = cz + ux * sn + uz * cs;
      const lp = live[live.length - 1];
      if (!lp || Math.hypot(coil.x - lp.x, coil.z - lp.z) >= SPACING) live.push({ x: coil.x, z: coil.z, y: landY(coil.x, coil.z), age: 0, sunk: 0 });
    }
    // the pile: what stands on each point, then each point rests on the pool (sinking) or on the loop beneath it
    const n = live.length, r2w = (1.7 * AL) ** 2;
    for (let i = 0; i < n; i++) {
      const p = live[i]; let load = 0;
      for (let j = i + NEIGH; j < n; j++) { const q = live[j]; if ((q.x - p.x) ** 2 + (q.z - p.z) ** 2 < r2w) load = Math.max(load, q.y - p.y); }
      if ((coil.x - p.x) ** 2 + (coil.z - p.z) ** 2 < r2w && n - i > NEIGH) load = Math.max(load, contactY - p.y);
      p.load = load;
    }
    if (topple.t >= 0) {                               // toppling: the upper loops tip over about the base of the pile
      topple.t += dt;
      if (topple.t < 0.9) for (const p of live) {
        const up = p.y - poolAt(p.x, p.z);
        if (up > 0.5 * AL && Math.hypot(p.x - coil.cx, p.z - coil.cz) < 3 * RC0) { const v = 1.5 * up * sstep(0, 0.3, topple.t); p.x += topple.dx * v * dt; p.z += topple.dz * v * dt; }
      }
      if (topple.t > 5) { topple.t = -1; nextTopple = simT + 9 + 9 * hash1(Math.floor(simT) + 0.71); }
    }
    for (let i = 0; i < n; i++) {
      const p = live[i]; p.age += dt;
      const soft = ropeAt(p.x, p.z), pool = sampleB(p.x, p.z) + soft + AL - p.sunk, sup = loopAt(p.x, p.z, i - NEIGH + 1);
      p.onPool = pool >= sup;
      if (p.onPool) p.sunk = Math.min(p.sunk + (SINK0 + KW * p.load + KS * soft) * dt, 1.8 * AL + soft);
      p.y = Math.min(p.y + VUP * dt, Math.max(p.y - VFALL * dt, Math.max(pool, sup)));
    }
    // a rope drapes over one it crosses rather than kinking up onto it: limit the slope between neighbouring points
    const sl = 0.6 * SPACING;
    for (let i = 1; i < n; i++) live[i].y = Math.max(live[i].y, live[i - 1].y - sl);
    for (let i = n - 2; i >= 0; i--) live[i].y = Math.max(live[i].y, live[i + 1].y - sl);
    // the oldest rope melts once it has sunk by its radius (or the chain grows too long, or it has lingered too long)
    nDep = 0;
    while (live.length > 2 && nDep < 8 && ((live[0].onPool && live[0].y - sampleB(live[0].x, live[0].z) <= -0.4 * AL) || live.length > NLMAX - 4 || live[0].age > 12)) {
      const a = live[0], b = live[1], len = Math.hypot(b.x - a.x, b.z - a.z), yc = (a.y + b.y) / 2, mx = (a.x + b.x) / 2, mz = (a.z + b.z) / 2;
      const rel = yc - sampleB(mx, mz), f = Math.max(0, Math.min(1, (rel + AL) / (2 * AL))), vol = Math.PI * AL * AL * len;
      DEP.set([a.x, a.z, b.x, b.z], nDep * 4); DEPY[nDep * 2] = AL; DEPY[nDep * 2 + 1] = yc; nDep++;
      splatB(mx, mz, vol * (1 - f));
      rsegs.push({ x0: a.x, z0: a.z, x1: b.x, z1: b.z, rel, vol: vol * f, age: 0 });
      live.shift();
    }
    // melted rope fades on the GPU at the same rate; the volume it loses goes into the mound
    const k = Math.exp(-dt / TAU);
    for (let i = rsegs.length - 1; i >= 0; i--) {
      const s = rsegs[i], e0 = Math.exp(-s.age / TAU);
      s.age += dt;
      const gone = s.age > 5 * TAU;
      splatB((s.x0 + s.x1) / 2, (s.z0 + s.z1) / 2, gone ? s.vol * e0 : s.vol * e0 * (1 - k));
      if (gone) rsegs.splice(i, 1);
    }
    stepB(dt);
    contactY += (landY(coil.x, coil.z) - contactY) * Math.min(1, dt * 25);
    pileH = contactY - poolAt(coil.cx, coil.cz);
    return k;
  }
  // the GPU side of one step: blur and fade the melted rope, then melt in the loops that have sunk
  let cur = 0;
  function gpuStep(k) {
    gl.viewport(0, 0, NR, NR);
    let u = (gl.useProgram(P.blur.p), P.blur.u);
    gl.bindFramebuffer(gl.FRAMEBUFFER, rF[1 - cur]); bind(u.S, 0, rT[cur]); gl.uniform2f(u.DIR, 1, 0); gl.uniform1f(u.K, k); gl.uniform1f(u.BL, 0.35); quad();
    gl.bindFramebuffer(gl.FRAMEBUFFER, rF[cur]); bind(u.S, 0, rT[1 - cur]); gl.uniform2f(u.DIR, 0, 1); gl.uniform1f(u.K, 1); quad();
    u = (gl.useProgram(P.dep.p), P.dep.u);
    gl.bindFramebuffer(gl.FRAMEBUFFER, rF[1 - cur]); bind(u.S, 0, rT[cur]); bind(u.TB, 1, texB);
    gl.uniform4fv(u.SEG, DEP); gl.uniform2fv(u.SY, DEPY); gl.uniform1i(u.NS, nDep); gl.uniform1f(u.LR, LR); gl.uniform1f(u.LB, LB); quad();
    cur = 1 - cur;
    gl.bindFramebuffer(gl.FRAMEBUFFER, null);
  }
  const uploadB = () => { gl.bindTexture(gl.TEXTURE_2D, texB); gl.texSubImage2D(gl.TEXTURE_2D, 0, 0, 0, NB, NB, gl.RED, gl.FLOAT, B); };

  // ---------------------------------------------------------------- camera: low and close, a macro lens
  const FOC0 = 4.5, EL = 0.47, DIST = 125, TGT = [0, 9, -7];   // a high three-quarter view into the bowl
  let FOC = FOC0;
  const cam = { p: [TGT[0], TGT[1] + Math.sin(EL) * DIST, TGT[2] + Math.cos(EL) * DIST], f: [0, -Math.sin(EL), -Math.cos(EL)], r: [1, 0, 0], u: [0, Math.cos(EL), -Math.sin(EL)] };
  let W = 1, H = 1;
  const resize = () => { const w = stage.clientWidth || 1, h = stage.clientHeight || 1, d = Math.min(devicePixelRatio, 2, Math.max(0.5, Math.sqrt(580000 / (w * h)))); W = canvas.width = Math.max(1, (w * d) | 0); H = canvas.height = Math.max(1, (h * d) | 0); FOC = FOC0 * Math.min(1, w / h / 1.15); dirty = true; };
  let dirty = true;
  const ro = new ResizeObserver(resize); ro.observe(stage); resize();
  const onMove = e => { const b = canvas.getBoundingClientRect(); px = (e.clientX - b.left) / b.width; py = (e.clientY - b.top) / b.height; lastMove = performance.now(); held = e.pointerType === 'mouse'; };
  const onLeave = () => { held = false; };
  canvas.addEventListener('pointermove', onMove); canvas.addEventListener('pointerdown', onMove); canvas.addEventListener('pointerleave', onLeave);
  const pointerOnPool = () => {
    const a = W / H, sx = (px - 0.5) * a, sy = 0.5 - py, d = cam.f.map((f, i) => f * FOC + cam.r[i] * sx + cam.u[i] * sy);
    if (d[1] > -1e-3) return [0, 0];
    const t = (H0 + 2 - cam.p[1]) / d[1];
    return [Math.max(-11, Math.min(11, cam.p[0] + d[0] * t)), Math.max(-11, Math.min(9, cam.p[2] + d[2] * t))];
  };

  // ---------------------------------------------------------------- frame
  // The falling thread: straight and vertical under the nozzle (gravity), then in the bottom "coiling" part, a few coil
  // radii high, it winds out from the axis into a helix that meets the contact point tangent to the loop being laid.
  // Dragged, the coil lags the nozzle and the thread leans between them, still vertical at both ends.
  const NBZ = 14, NHX = 8, BZU = new Float32Array(NBZ * 4);
  function thread() {
    let ox = coil.x - coil.cx, oz = coil.z - coil.cz;
    const rc = Math.hypot(ox, oz) || 1e-3, hc = 1.3 * rc + 1.6 + 0.6 * AL, yA = contactY + hc;
    const dth = 2.4 * (1 - 0.75 * coil.ecc), dir = coil.dir;
    let cx = 0, cy = 0, cz = 0, rad = 0;
    for (let i = 0; i < NHX; i++) {                // the helix, from the contact (s = 0) up to the axis (s = 1)
      const s = i / (NHX - 1), a = dir * dth * s, w = (1 - s) * (1 - s) * (1 + 2 * s), c = Math.cos(a), sn = Math.sin(a);
      const x = coil.cx + (ox * c - oz * sn) * w, z = coil.cz + (ox * sn + oz * c) * w, y = contactY + hc * (0.3 * s + 0.7 * s * s);
      const r = A1 + (AL - A1) * (1 - sstep(0, 1, s));
      BZU.set([x, y, z, r], (NHX - 1 - i) * 4);
    }
    const lean = Math.hypot(noz.x - coil.cx, noz.z - coil.cz), yb = yA + 2.5 + 1.5 * lean;
    for (let i = NHX; i < NBZ; i++) {              // the leaning stretch above it, up to the nozzle's plumb line
      const t = (i - NHX + 1) / (NBZ - NHX), e = t * t * (3 - 2 * t), y = yA + (yb - yA) * t;
      BZU.set([coil.cx + (noz.x - coil.cx) * e, y, coil.cz + (noz.z - coil.cz) * e, A1 * Math.pow(1 + (y - yA) / 9, 0.42)], i * 4);
    }
    // BZU runs top-down from the axis: reorder so index 0 is the top (nozzle side) and NBZ-1 the contact
    const tmp = Array.from({ length: NBZ }, (_, i) => Array.from(BZU.subarray(i * 4, i * 4 + 4)));
    const order = [...tmp.slice(NHX).reverse(), ...tmp.slice(0, NHX)];
    order.forEach((v, i) => BZU.set(v, i * 4));
    for (let i = 0; i < NBZ; i++) { cx += BZU[i * 4] / NBZ; cy += BZU[i * 4 + 1] / NBZ; cz += BZU[i * 4 + 2] / NBZ; }
    for (let i = 0; i < NBZ; i++) rad = Math.max(rad, Math.hypot(BZU[i * 4] - cx, BZU[i * 4 + 1] - cy, BZU[i * 4 + 2] - cz) + BZU[i * 4 + 3]);
    return { yb, yA, bs: [cx, cy, cz, rad + 0.3] };
  }
  const LVU = new Float32Array(NLMAX * 4), LGU = new Float32Array(NLMAX);
  function packLive() {
    const pts = live.slice(-(NLMAX - 1)); pts.push({ x: coil.x, y: contactY, z: coil.z });
    const n = pts.length; let cx = 0, cy = 0, cz = 0, rad = 0;
    pts.forEach((p, i) => { LVU.set([p.x, p.y, p.z, AL], i * 4); cx += p.x / n; cy += p.y / n; cz += p.z / n; });
    pts.forEach(p => { rad = Math.max(rad, Math.hypot(p.x - cx, p.y - cy, p.z - cz)); });
    for (let g = 0; g < NLMAX / 4; g++) {
      const i0 = g * 4, i1 = Math.min(n - 1, i0 + 4); if (i0 >= n - 1) { LGU.set([0, -1e4, 0, 0], g * 4); continue; }
      let gx = 0, gy = 0, gz = 0, gr = 0; for (let i = i0; i <= i1; i++) { gx += pts[i].x; gy += pts[i].y; gz += pts[i].z; }
      const m = i1 - i0 + 1; gx /= m; gy /= m; gz /= m;
      for (let i = i0; i <= i1; i++) gr = Math.max(gr, Math.hypot(pts[i].x - gx, pts[i].y - gy, pts[i].z - gz));
      LGU.set([gx, gy, gz, gr + AL + 0.05], g * 4);
    }
    let sh = 0; pts.forEach(p => { sh += Math.max(0, p.y + AL - sampleB(p.x, p.z)) / n; });
    return { n, ls: [cx, cy, cz, rad + AL + 0.2], stk: [cx, cz, Math.max(1.5, Math.hypot(coil.x - cx, coil.z - cz) + AL), sh * 0.45] };
  }
  function draw() {
    const bz = thread(), lv = packLive();
    uploadB();
    const u = (gl.useProgram(P.main.p), P.main.u);
    gl.viewport(0, 0, W, H);
    bind(u.TB, 0, texB); bind(u.TR, 1, rT[cur]);
    gl.uniform4fv(u.BZ, BZU); gl.uniform4fv(u.BS, bz.bs); gl.uniform3f(u.NZ, noz.x, bz.yb, noz.z); gl.uniform1f(u.YA, bz.yA);
    gl.uniform4fv(u.LV, LVU); gl.uniform4fv(u.LG, LGU); gl.uniform1i(u.NLV, lv.n); gl.uniform4fv(u.LS, lv.ls);
    gl.uniform1f(u.A1, A1); gl.uniform1f(u.HMAX, Math.max(maxB, contactY + AL) + AL * 1.2 + 0.4); gl.uniform1f(u.H0, H0);
    gl.uniform1f(u.LB, LB); gl.uniform1f(u.LR, LR); gl.uniform1f(u.EXPO, 0.8);
    gl.uniform2f(u.CC, coil.x, coil.z); gl.uniform1f(u.CA, AL);
    const hl = (sampleB(0, -21) + sampleB(21, 0) + sampleB(-21, 0) + sampleB(0, 21)) / 4, rs = 30 * Math.sqrt(Math.max(0, 1 - ((10 - hl) / 10) ** 2));
    gl.uniform1f(u.HL, hl); gl.uniform1f(u.RS, rs); gl.uniform4fv(u.STK, lv.stk);
    gl.uniform3fv(u.CP, cam.p); gl.uniform3fv(u.CF, cam.f); gl.uniform3fv(u.CR, cam.r); gl.uniform3fv(u.CU, cam.u); gl.uniform1f(u.FOC, FOC); gl.uniform2f(u.R, W, H);
    quad();
  }
  // start with a mound already built: a few seconds of pouring before the first frame
  for (let i = 0; i < 300; i++) { gpuStep(advance(1 / 30, false, null)); if (i % 10 === 9) uploadB(); }
  let raf, last = performance.now();
  function frame(now) {
    const dt = Math.min(1 / 30, Math.max(0, (now - last) / 1000)); last = now;
    if (!reduce) {
      const live = held || now - lastMove < 1800;
      gpuStep(advance(dt, live, live ? pointerOnPool() : null));
      draw();
    } else if (dirty) { draw(); dirty = false; }
    raf = requestAnimationFrame(frame);
  }
  raf = requestAnimationFrame(frame);
  return () => {
    cancelAnimationFrame(raf); ro.disconnect();
    canvas.removeEventListener('pointermove', onMove); canvas.removeEventListener('pointerdown', onMove); canvas.removeEventListener('pointerleave', onLeave);
    gl.getExtension('WEBGL_lose_context')?.loseContext(); canvas.remove();
  };
}

Lightbox zoom

Lightbox zoom is a DOM effect: the square crop opens out to the full photo. One plain JavaScript file with no dependencies; add it with npx motion-library add lightbox-zoom.

Open it live, tune it and copy the code

What is Lightbox zoom?

A tapped photo grows from its square into a full-screen viewer; swipe down and it shrinks back home.

How does Lightbox zoom work?

The tile is a centre crop, so the photo scales uniformly while a clip-path opens the crop to the full frame: FLIP with only transform and clip-path.

How do I add Lightbox zoom to my website?

  1. In your project folder, run npx motion-library add lightbox-zoom. It writes animations/lightbox-zoom.js. Needs Node 18 or newer.
  2. Import it and call it with a container element that has a size: import fx from './animations/lightbox-zoom.js'; const stop = fx(document.querySelector('#hero'));
  3. Call stop() to clean up, for example when a component unmounts.

Every setting below is also a flag, so you can install it already tuned: npx motion-library add lightbox-zoom --duration 1000.

// Lightbox zoom: tap a photo in your library and it grows out of its square into the full-screen viewer over a black backdrop; swipe it down and it shrinks back into the same square.
// The square tile is a centre crop of the full photo, so the photo scales uniformly while a clip opens the crop out to the full frame (FLIP: transform and clip-path only). The dismiss starts by following the finger down, then flies home, which is why the way back feels like your own gesture.
const PARAMS = { duration: 440, backdrop: 1 };

export default function (stage) {
  const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;
  let uid = 0;
  const tag = Math.random().toString(36).slice(2, 7);
  // A landscape photo drawn in SVG: a sky gradient, an optional sun, layered ridges and optional water.
  const photo = p => {
    const id = `lb${tag}${uid++}`, W = 120, H = 90;
    let s = `<svg viewBox="0 0 ${W} ${H}" preserveAspectRatio="xMidYMid slice"><defs><linearGradient id="${id}" x1="0" y1="0" x2="0" y2="1"><stop offset="0" stop-color="${p.sky[0]}"/><stop offset="1" stop-color="${p.sky[1]}"/></linearGradient></defs><rect width="${W}" height="${H}" fill="url(#${id})"/>`;
    if (p.sun) s += `<circle cx="${p.sun[0]}" cy="${p.sun[1]}" r="${p.sun[2]}" fill="${p.sun[3]}"/>`;
    p.ridges.forEach((c, i) => {
      const base = H * (p.y0 + i * p.dy), amp = p.amp * (1 - i * 0.22);
      let d = `M0 ${H}`;
      for (let x = 0; x <= W; x += 3) d += ` L${x} ${(base - amp * (Math.sin(x * 0.045 + p.seed + i * 1.7) * 0.6 + Math.sin(x * 0.11 + p.seed * 2 + i) * 0.3 + Math.sin(x * 0.23 + i * 3) * 0.1)).toFixed(1)}`;
      s += `<path d="${d} L${W} ${H}Z" fill="${c}"/>`;
    });
    if (p.water) s += `<rect y="${H * p.water[0]}" width="${W}" height="${H}" fill="${p.water[1]}"/><path d="M20 ${H * p.water[0] + 6}h22M66 ${H * p.water[0] + 10}h30M34 ${H * p.water[0] + 16}h18" stroke="#fff" stroke-opacity=".25" stroke-width="1.2"/>`;
    return s + '</svg>';
  };
  const P = [
    { sky: ['#f4b39a', '#fde4cc'], sun: [86, 42, 9, '#fff3df'], ridges: ['#c48f95', '#8f6a83', '#4d3f5e'], y0: 0.55, dy: 0.11, amp: 16, seed: 1 },
    { sky: ['#8fc3dc', '#e4f1f4'], ridges: ['#8db1bf', '#557f8e', '#2d4f5b'], water: [0.8, '#5b8796'], y0: 0.42, dy: 0.12, amp: 22, seed: 2 },

    { sky: ['#cbd8cd', '#eef2ea'], ridges: ['#a9bfaa', '#7a977d', '#46634e'], y0: 0.42, dy: 0.15, amp: 14, seed: 4 },
    { sky: ['#2f3a68', '#e08a6a'], sun: [62, 60, 8, '#ffd59a'], ridges: ['#28305a'], water: [0.66, '#34416f'], y0: 0.64, dy: 0, amp: 5, seed: 5 },
    { sky: ['#5d9de4', '#d8eafb'], ridges: ['#eef3f8', '#93a7bd', '#4f6378'], y0: 0.4, dy: 0.15, amp: 24, seed: 6 },
    { sky: ['#0d1430', '#2a3a6e'], sun: [92, 20, 5, '#f4f1e0'], ridges: ['#1b2546', '#10172e'], y0: 0.62, dy: 0.1, amp: 10, seed: 7 },
    { sky: ['#b9dcef', '#f2f8ef'], ridges: ['#b7d48c', '#88b462', '#5f8c43'], y0: 0.55, dy: 0.12, amp: 10, seed: 8 },
    { sky: ['#bfb0e3', '#f2e7f5'], sun: [40, 38, 8, '#fffaf4'], ridges: ['#a08bcb', '#7460a6', '#4a3d78'], y0: 0.52, dy: 0.12, amp: 14, seed: 9 },
  ];
  const tiles = Array.from({ length: 15 }, (_, i) => P[(i * 4 + (i >> 2)) % P.length]);
  const sbIcons = c => `<svg width="58" height="12" viewBox="0 0 58 12"><g fill="${c}"><rect x="0" y="8" width="3" height="4" rx=".8"/><rect x="4.5" y="6" width="3" height="6" rx=".8"/><rect x="9" y="3.5" width="3" height="8.5" rx=".8"/><rect x="13.5" y="1" width="3" height="11" rx=".8"/><path d="M26 3.6a8.5 8.5 0 0 1 11 0l-1.3 1.4a6.6 6.6 0 0 0-8.4 0zM28.4 6.3a5 5 0 0 1 6.2 0l-1.3 1.4a3 3 0 0 0-3.6 0zM31.5 11.5l-1.6-1.8a2.4 2.4 0 0 1 3.2 0z"/><rect x="40.5" y=".8" width="15" height="10.4" rx="3" fill="none" stroke="${c}" stroke-opacity=".45"/><rect x="42" y="2.3" width="10" height="7.4" rx="1.6"/></g></svg>`;
  const ico = d => `<svg viewBox="0 0 24 24" width="20" height="20" fill="none" stroke="#fff" stroke-width="1.7" stroke-linecap="round" stroke-linejoin="round">${d}</svg>`;
  const FW = 266, FH = 200, FY = 183; // the photo in the viewer: full width, 4:3, centred

  const root = document.createElement('div');
  root.className = 'lb-root';
  root.innerHTML = `<style>
    .lb-root { position: absolute; inset: 0; overflow: hidden; background: radial-gradient(120% 100% at 50% 0%, #1b1b1e, #0d0d0f); font-family: "Mona Sans", system-ui, -apple-system, sans-serif; color: #15171a; }

    .lb-sb { position: absolute; left: 0; right: 0; top: 0; height: 44px; padding: 0 24px 0 32px; display: flex; align-items: center; justify-content: space-between; font-size: 13px; font-weight: 650; z-index: 8; }
    .lb-sb.w { color: #fff; opacity: 0; }
    .lb-lib { position: absolute; inset: 0; padding-top: 44px; }
    .lb-top { display: flex; justify-content: space-between; align-items: baseline; margin: 6px 14px 2px; }
    .lb-top h2 { margin: 0; font-size: 27px; font-weight: 750; letter-spacing: -0.02em; }
    .lb-top span { font-size: 13px; color: #007aff; }
    .lb-day { margin: 0 14px 10px; font-size: 12.5px; color: #8a8a8f; }
    .lb-grid { display: grid; grid-template-columns: repeat(3, 1fr); gap: 2px; }
    .lb-grid div { aspect-ratio: 1; overflow: hidden; }
    .lb-grid svg, .lb-photo svg { display: block; width: 100%; height: 100%; }
    .lb-back { position: absolute; inset: 0; background: #000; opacity: 0; }
    .lb-photo { position: absolute; left: 0; top: ${FY}px; width: ${FW}px; height: ${FH}px; transform-origin: 0 0; visibility: hidden; will-change: transform, clip-path; }
    .lb-chrome { position: absolute; left: 0; right: 0; display: flex; align-items: center; justify-content: space-between; padding: 0 16px; color: #fff; opacity: 0; }
    .lb-chrome.t { top: 44px; height: 44px; }
    .lb-chrome.b { bottom: 22px; height: 44px; padding: 0 28px; }
    .lb-date { text-align: center; font-size: 13px; font-weight: 640; line-height: 1.25; }
    .lb-date small { display: block; font-size: 11px; font-weight: 500; opacity: .7; }

  </style>
  <div class="lb-phone"><div class="lb-screen">
    <div class="lb-lib"><div class="lb-top"><h2>Library</h2><span>Select</span></div><div class="lb-day">Today · Lofoten, Norway</div>
      <div class="lb-grid">${tiles.map(p => `<div>${photo(p)}</div>`).join('')}</div></div>
    <div class="lb-back"></div>
    <div class="lb-photo"></div>
    <div class="lb-chrome t">${ico('<path d="M15 5l-7 7 7 7"/>')}<div class="lb-date">14 August<small>18:42</small></div><span style="width:20px"></span></div>
    <div class="lb-chrome b">${ico('<path d="M12 15V4M8 8l4-4 4 4M6 11v8.5h12V11"/>')}${ico('<path d="M12 20s-7-4.4-7-10a4 4 0 0 1 7-2.6A4 4 0 0 1 19 10c0 5.6-7 10-7 10z"/>')}${ico('<circle cx="12" cy="12" r="8"/><path d="M12 11v5M12 8h.01"/>')}${ico('<path d="M5 7h14M10 7V5h4v2M7 7l1 13h8l1-13"/>')}</div>
    <div class="lb-sb"><span>9:41</span>${sbIcons('#15171a')}</div><div class="lb-sb w"><span>9:41</span>${sbIcons('#fff')}</div>
    <div class="lb-tap"></div>
  </div><div class="lb-island"></div></div>`;
  stage.append(root);
  const $ = s => root.querySelector(s);
  const phone = $('.lb-phone'), screen = $('.lb-screen'), lib = $('.lb-lib'), backdrop = $('.lb-back'), ph = $('.lb-photo'), sbW = $('.lb-sb.w'), tapEl = $('.lb-tap');
  const chrome = [...root.querySelectorAll('.lb-chrome')], cells = [...root.querySelectorAll('.lb-grid div')];

  const fit = () => {
    const W = stage.clientWidth, H = stage.clientHeight, f = Math.min(1, Math.max(0.66, 0.66 + (H - 260) / 240 * 0.34));
    const s = Math.min(H * 0.92 / (580 * f), W * 0.86 / 280), top = f >= 1 ? (H - 580 * s) / 2 : H * 0.05;
    phone.style.transform = `translate(-50%, ${top}px) scale(${s})`;
  };
  const ro = new ResizeObserver(fit); ro.observe(stage); fit();

  const local = el => { const sr = screen.getBoundingClientRect(), r = el.getBoundingClientRect(), k = sr.width / 266; return { x: (r.left - sr.left) / k, y: (r.top - sr.top) / k, w: r.width / k, h: r.height / k }; };
  // The square tile shows the middle of the photo: scale uniformly and clip the sides away.
  const inverted = t => {
    const k = Math.max(t.w / FW, t.h / FH), ix = (FW - t.w / k) / 2, iy = (FH - t.h / k) / 2;
    return { transform: `translate(${t.x - k * ix}px, ${t.y - FY - k * iy}px) scale(${k})`, clipPath: `inset(${iy}px ${ix}px ${iy}px ${ix}px)` };
  };
  const full = { transform: 'translate(0px, 0px) scale(1)', clipPath: 'inset(0px 0px 0px 0px)' };
  const dragged = { transform: `translate(${FW * 0.05}px, ${FH * 0.05 + 54}px) scale(0.9)`, clipPath: 'inset(0px 0px 0px 0px)' };
  const easeOut = 'cubic-bezier(0.23, 1, 0.32, 1)', curve = 'cubic-bezier(0.32, 0.72, 0, 1)';
  const tapAt = (x, y, dy = 0) => {
    if (reduce) dy = 0;
    tapEl.style.left = x + 'px'; tapEl.style.top = y + 'px';
    tapEl.animate(dy ? [{ opacity: 0, transform: reduce ? 'none' : 'scale(0.8)' }, { opacity: 1, transform: 'scale(1)', offset: 0.15 }, { opacity: 1, transform: `translateY(${dy}px)`, offset: 0.85 }, { opacity: 0, transform: `translateY(${dy}px) scale(1.1)` }]
      : [{ opacity: 0, transform: reduce ? 'none' : 'scale(0.8)' }, { opacity: 1, transform: 'scale(1)', offset: 0.3 }, { opacity: 0, transform: reduce ? 'none' : 'scale(1.2)' }], { duration: dy ? 520 : 460, easing: easeOut });
  };

  function open(i) {
    const cell = cells[i], t = local(cell);
    ph.innerHTML = cell.innerHTML; ph.style.visibility = 'visible'; cell.style.visibility = 'hidden';
    const o = { duration: PARAMS.duration, easing: curve, fill: 'both' };
    // Reduced motion: the photo fades in full size instead of zooming out of its square.
    ph.animate(reduce ? [{ ...full, opacity: 0 }, { ...full, opacity: 1 }] : [inverted(t), full], o);
    backdrop.animate([{ opacity: 0 }, { opacity: PARAMS.backdrop }], { duration: PARAMS.duration * 0.7, easing: easeOut, fill: 'both' });
    sbW.animate([{ opacity: 0 }, { opacity: 1 }], { duration: 200, delay: 80, fill: 'both' });
    chrome.forEach(c => c.animate([{ opacity: 0 }, { opacity: 1 }], { duration: 200, delay: PARAMS.duration * 0.6, easing: easeOut, fill: 'both' }));
  }
  function dismiss(i) {
    const cell = cells[i], t = local(cell);
    // 1. The photo follows the finger down and shrinks a little; the black thins out.
    chrome.forEach(c => c.animate([{ opacity: 1 }, { opacity: 0 }], { duration: 120, easing: easeOut, fill: 'both' }));
    ph.animate([full, reduce ? full : dragged], { duration: 300, easing: easeOut, fill: 'both' });
    backdrop.animate([{ opacity: PARAMS.backdrop }, { opacity: PARAMS.backdrop * 0.55 }], { duration: 300, easing: easeOut, fill: 'both' });
    // 2. Release: it flies back into its square.
    later(() => {
      if (reduce) cell.style.visibility = ''; // the square is already back under the fading photo
      ph.animate(reduce ? [{ ...full, opacity: 1 }, { ...full, opacity: 0 }] : [dragged, inverted(t)], { duration: PARAMS.duration, easing: curve, fill: 'both' }).onfinish = () => { cell.style.visibility = ''; ph.style.visibility = 'hidden'; };
      backdrop.animate([{ opacity: PARAMS.backdrop * 0.55 }, { opacity: 0 }], { duration: PARAMS.duration * 0.6, easing: easeOut, fill: 'both' });
      sbW.animate([{ opacity: 1 }, { opacity: 0 }], { duration: 200, fill: 'both' });
    }, 320);
  }

  const timers = new Set();
  const later = (fn, ms) => { const id = setTimeout(() => { timers.delete(id); fn(); }, ms); timers.add(id); };
  const order = [4, 9, 1, 12];
  let n = 0;
  function cycle() {
    const i = order[n++ % order.length], t = local(cells[i]);
    later(() => tapAt(t.x + t.w / 2, t.y + t.h / 2), 700);
    later(() => open(i), 820);
    later(() => tapAt(FW / 2, FY + FH / 2, 54), 820 + PARAMS.duration + 1600);
    later(() => dismiss(i), 820 + PARAMS.duration + 1680);
    later(cycle, 820 + PARAMS.duration * 2 + 1680 + 320 + 900);
  }
  cycle();

  return () => { timers.forEach(clearTimeout); ro.disconnect(); root.getAnimations({ subtree: true }).forEach(a => a.cancel()); root.remove(); };
}

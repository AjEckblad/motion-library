Roche limit

Roche limit is a WebGL effect: tidal stretching past the Roche limit shears rubble into a ring. One plain JavaScript file with no dependencies; add it with npx motion-library add roche-limit.

Open it live, tune it and copy the code

What is Roche limit?

A rubble moon spirals into an ice giant, tears into two streams and winds into a ring.

How does Roche limit work?

Twenty thousand particles feel the moon’s self-gravity and the planet’s tide, which grows as 1/d³; inner rubble orbits faster, shearing the streams into a ring.

How do I add Roche limit to my website?

  1. In your project folder, run npx motion-library add roche-limit. It writes animations/roche-limit.js. Needs Node 18 or newer.
  2. Import it and call it with a container element that has a size: import fx from './animations/roche-limit.js'; const stop = fx(document.querySelector('#hero'));
  3. Call stop() to clean up, for example when a component unmounts.

Every setting below is also a flag, so you can install it already tuned: npx motion-library add roche-limit --particles 40000.

// Roche limit: a rubble-pile moon spirals in toward an ice giant. Its own gravity holds it together, but the planet
// pulls harder on the near side than the far side, and that difference (the tide) grows as 1/d³. Inside about 2.4
// planet radii the tide wins: the moon stretches, then sheds two streams, one inward and one outward, from the points
// where the pull balances. Inner rubble orbits faster than outer rubble, so the streams shear and wind round the
// planet until they close into a ring. Twenty thousand particles, each lit by the sun and shadowed by the planet.
const PARAMS = { particles: 20000, speed: 1, glow: 1 };

export default function (stage) {
  const canvas = document.createElement('canvas');
  canvas.style.cssText = 'position:absolute;inset:0;width:100%;height:100%;display:block';
  stage.append(canvas);
  const gl = canvas.getContext('webgl2', { antialias: false, premultipliedAlpha: false });
  if (!gl) { stage.textContent = 'This needs WebGL2.'; return () => canvas.remove(); }
  const N = Math.max(2000, Math.round(PARAMS.particles));
  const SUN = (() => { const v = [0.8, 0.32, 0.35], l = Math.hypot(...v); return v.map(x => x / l); })();

  // ── the simulation (planet radius 1, GM = 1)
  const px = new Float32Array(N), py = new Float32Array(N), pz = new Float32Array(N), vx = new Float32Array(N), vy = new Float32Array(N), vz = new Float32Array(N);
  const bound = new Uint8Array(N), lit = new Float32Array(N);
  const RM = 0.24, KB = 2 / 2.4 ** 3;                            // moon radius; self-gravity strength that fails at d ≈ 2.4
  let com = [0, 0, 0], comV = [0, 0, 0], simT = 0, cycleT = 0;
  function reset() {
    const d0 = 3.5, v0 = Math.sqrt(1 / d0);
    for (let i = 0; i < N; i++) {
      let x, y, z; do { x = Math.random() * 2 - 1; y = Math.random() * 2 - 1; z = Math.random() * 2 - 1; } while (x * x + y * y + z * z > 1);
      px[i] = d0 + x * RM; py[i] = y * RM; pz[i] = z * RM; vx[i] = 0; vy[i] = 0; vz[i] = v0; bound[i] = 1;
    }
    simT = 0; cycleT = 0;
  }
  reset();
  function step(h) {
    // the clump that is still a moon: its centre and velocity
    let n = 0, cx = 0, cy = 0, cz = 0, cvx = 0, cvy = 0, cvz = 0;
    for (let i = 0; i < N; i++) if (bound[i]) { n++; cx += px[i]; cy += py[i]; cz += pz[i]; cvx += vx[i]; cvy += vy[i]; cvz += vz[i]; }
    if (n) { com = [cx / n, cy / n, cz / n]; comV = [cvx / n, cvy / n, cvz / n]; }
    const decay = 0.0032;                                       // the moon's orbit decays (tides raised on the planet)
    for (let i = 0; i < N; i++) {
      const r2 = px[i] * px[i] + py[i] * py[i] + pz[i] * pz[i], r = Math.sqrt(r2), g = -1 / (r2 * r);
      let ax = px[i] * g, ay = py[i] * g, az = pz[i] * g;
      if (bound[i]) {
        const dx = px[i] - com[0], dy = py[i] - com[1], dz = pz[i] - com[2];
        if (dx * dx + dy * dy + dz * dz > (RM * 1.5) ** 2) bound[i] = 0;   // pulled too far from the clump: free rubble now
        else {
          ax -= KB * dx + 0.3 * (vx[i] - comV[0]); ay -= KB * dy + 0.3 * (vy[i] - comV[1]); az -= KB * dz + 0.3 * (vz[i] - comV[2]);   // self-gravity, and friction inside the pile
          ax -= decay * vx[i]; ay -= decay * vy[i]; az -= decay * vz[i];
        }
      } else {
        // free rubble collides and settles: radial and vertical motion are slowly damped, so the ring turns round and flat
        const vr = (vx[i] * px[i] + vy[i] * py[i] + vz[i] * pz[i]) / r;
        const kick = (Math.random() - 0.5) * 0.05;                  // collisions also jostle them, spreading the streams into a band
        ax += (kick - 0.03 * vr) * px[i] / r; ay -= 0.03 * vr * py[i] / r + 0.04 * vy[i]; az += (kick - 0.03 * vr) * pz[i] / r;
      }
      vx[i] += ax * h; vy[i] += ay * h; vz[i] += az * h;
      px[i] += vx[i] * h; py[i] += vy[i] * h; pz[i] += vz[i] * h;
      if (r < 1.02) { const k = 1.02 / r; px[i] *= k; py[i] *= k; pz[i] *= k; vx[i] *= 0.5; vy[i] *= 0.5; vz[i] *= 0.5; }   // grazes the cloud tops
    }
  }

  // ── GL
  const sh = (type, src) => { const s = gl.createShader(type); gl.shaderSource(s, src); gl.compileShader(s); if (!gl.getShaderParameter(s, gl.COMPILE_STATUS)) throw new Error(gl.getShaderInfoLog(s)); return s; };
  const prog = (vs, fs) => {
    const p = gl.createProgram(); gl.attachShader(p, sh(gl.VERTEX_SHADER, vs)); gl.attachShader(p, sh(gl.FRAGMENT_SHADER, fs)); gl.bindAttribLocation(p, 0, 'p'); gl.linkProgram(p);
    if (!gl.getProgramParameter(p, gl.LINK_STATUS)) throw new Error(gl.getProgramInfoLog(p));
    const u = {}; for (let i = 0, n = gl.getProgramParameter(p, gl.ACTIVE_UNIFORMS); i < n; i++) { const nm = gl.getActiveUniform(p, i).name; u[nm] = gl.getUniformLocation(p, nm); }
    return { p, u };
  };
  const CAMERA = `uniform vec3 CP, CR, CU, CF, SUN; uniform float FOC, A;`;
  // the planet and the stars, per pixel: a ray against a sphere, banded clouds, a soft terminator, a blue rim
  const planetP = prog(`#version 300 es
  in vec2 p; out vec2 v; void main() { v = p; gl_Position = vec4(p, 0, 1); }`, `#version 300 es
  precision highp float; in vec2 v; out vec4 o; ${CAMERA} uniform float T, RING;
  float h(vec2 p) { return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453); }
  float n2(vec2 p) { vec2 i = floor(p), f = fract(p); f = f * f * (3. - 2. * f); return mix(mix(h(i), h(i + vec2(1, 0)), f.x), mix(h(i + vec2(0, 1)), h(i + 1.), f.x), f.y); }
  void main() {
    vec3 d = normalize(CF * FOC + CR * v.x * A + CU * v.y);
    float b = dot(CP, d), c = dot(CP, CP) - 1., disc = b * b - c;
    vec3 col = vec3(0);
    vec2 sp = vec2(atan(d.z, d.x), asin(d.y)) * 230., ci = floor(sp);
    vec3 p3 = fract(vec3(ci.xyx) * .1031); p3 += dot(p3, p3.yzx + 33.33); float s = fract((p3.x + p3.y) * p3.z);
    col += vec3(.8, .85, 1.) * step(.992, s) * (s - .992) * 70. * smoothstep(.45, .1, length(fract(sp) - .5));
    if (disc > 0.) {
      float t = -b - sqrt(disc); vec3 pos = CP + d * t, n = normalize(pos);
      float lat = n.y, lon = atan(n.z, n.x);
      float turb = n2(vec2(lon * 3. + T * .02, lat * 14.)) * .6 + n2(vec2(lon * 9. - T * .03, lat * 40.)) * .25;
      float band = sin(lat * 15. + turb * 3.);
      vec3 base = mix(vec3(.28, .52, .66), vec3(.55, .78, .84), .5 + .5 * band);
      base = mix(base, vec3(.85, .93, .95), smoothstep(.6, 1., band) * .35);
      float dl = dot(n, SUN), lambert = smoothstep(-.12, .55, dl);
      // the ring's shadow on the cloud tops, once there is a ring
      vec3 toS = SUN; float tr = -pos.y / toS.y; vec3 hit = pos + toS * tr; float rr = length(hit.xz);
      float shade = 1. - RING * .55 * step(0., tr) * smoothstep(1.35, 1.5, rr) * smoothstep(3.2, 2.7, rr);
      col = base * (.035 + .965 * lambert) * shade + base * .03 * RING;   // a little ring-shine on the night side
      col += vec3(.35, .6, 1.) * pow(1. - max(dot(n, -d), 0.), 3.) * (.25 + .75 * smoothstep(-.2, .4, dl)) * .8;   // atmosphere at the limb
    }
    o = vec4(col, 1);
  }`);
  // the rubble: points, hidden when the planet is in the way, dark in its shadow, lit by the sun
  const rockP = prog(`#version 300 es
  layout(location = 0) in vec3 P; layout(location = 1) in float L; ${CAMERA} uniform float PS; out float b;
  void main() {
    vec3 rel = P - CP; float dist = length(rel); vec3 d = rel / dist;
    float bb = dot(CP, d), cc = dot(CP, CP) - 1., disc = bb * bb - cc, t = -bb - sqrt(max(disc, 0.));
    float hidden = disc > 0. && t < dist ? 0. : 1.;
    float sd = dot(P, SUN); vec3 perp = P - SUN * sd; float shadow = sd < 0. && dot(perp, perp) < 1. ? .06 : 1.;
    vec3 c = vec3(dot(rel, CR), dot(rel, CU), dot(rel, CF));
    gl_Position = vec4(c.x * FOC / (c.z * A), c.y * FOC / c.z, 0, 1);
    gl_PointSize = PS * 6. / max(c.z, .5);
    b = hidden * shadow * L;
  }`, `#version 300 es
  precision mediump float; in float b; out vec4 o; uniform float K;
  void main() { vec2 q = gl_PointCoord * 2. - 1.; float f = max(0., 1. - dot(q, q)); o = vec4(vec3(1., .96, .9) * b * f * K, 1); }`);

  const quad = gl.createVertexArray(); gl.bindVertexArray(quad);
  gl.bindBuffer(gl.ARRAY_BUFFER, gl.createBuffer()); gl.bufferData(gl.ARRAY_BUFFER, new Float32Array([-1, -1, 3, -1, -1, 3]), gl.STATIC_DRAW);
  gl.enableVertexAttribArray(0); gl.vertexAttribPointer(0, 2, gl.FLOAT, false, 0, 0);
  const rv = gl.createVertexArray(); gl.bindVertexArray(rv);
  const pos = new Float32Array(N * 3), pb = gl.createBuffer(), lb = gl.createBuffer();
  gl.bindBuffer(gl.ARRAY_BUFFER, pb); gl.bufferData(gl.ARRAY_BUFFER, pos.byteLength, gl.DYNAMIC_DRAW); gl.enableVertexAttribArray(0); gl.vertexAttribPointer(0, 3, gl.FLOAT, false, 0, 0);
  gl.bindBuffer(gl.ARRAY_BUFFER, lb); gl.bufferData(gl.ARRAY_BUFFER, lit.byteLength, gl.DYNAMIC_DRAW); gl.enableVertexAttribArray(1); gl.vertexAttribPointer(1, 1, gl.FLOAT, false, 0, 0);
  gl.bindVertexArray(null);

  let W = 1, H = 1;
  const resize = () => { const d = Math.min(devicePixelRatio, 2); W = canvas.width = Math.max(1, (stage.clientWidth * d) | 0); H = canvas.height = Math.max(1, (stage.clientHeight * d) | 0); };
  const ro = new ResizeObserver(resize); ro.observe(stage); resize();

  let raf, last = performance.now(), t = 0;
  function frame(now) {
    const dt = Math.min(0.05, (now - last) / 1000); last = now; t += dt;
    const simDt = dt * 6.5 * PARAMS.speed;
    for (let s = 0; s < 3; s++) step(simDt / 3);
    simT += simDt; cycleT += dt * PARAMS.speed;
    // the story repeats: once the ring has closed and settled for a while, a fresh moon arrives
    const fade = Math.min(1, cycleT / 1.2) * Math.min(1, Math.max(0, (52 - cycleT) / 2));
    if (cycleT > 52) reset();
    // light each particle: the moon is a lit ball (day side and night side); loose rubble just reflects
    let free = 0;
    for (let i = 0; i < N; i++) {
      pos[i * 3] = px[i]; pos[i * 3 + 1] = py[i]; pos[i * 3 + 2] = pz[i];
      if (bound[i]) { const dx = px[i] - com[0], dy = py[i] - com[1], dz = pz[i] - com[2], l = Math.hypot(dx, dy, dz) + 1e-6; lit[i] = (0.12 + 0.88 * Math.max(0, (dx * SUN[0] + dy * SUN[1] + dz * SUN[2]) / l)) * fade * 0.4; }
      else { lit[i] = fade; free++; }
    }
    // camera: circling slowly, a little above the ring plane
    const az = 0.6 + t * 0.05, el = 0.26, dist = 5.4;
    const CP = [Math.cos(az) * Math.cos(el) * dist, Math.sin(el) * dist, Math.sin(az) * Math.cos(el) * dist];
    let CF = CP.map(v => -v / dist), CR = [CF[2], 0, -CF[0]]; const rl = Math.hypot(CR[0], CR[2]); CR = CR.map(v => v / rl);
    const CU = [CF[1] * CR[2] - CF[2] * CR[1], CF[2] * CR[0] - CF[0] * CR[2], CF[0] * CR[1] - CF[1] * CR[0]];
    const set = P => { gl.uniform3f(P.u.CP, ...CP); gl.uniform3f(P.u.CR, ...CR); gl.uniform3f(P.u.CU, ...CU); gl.uniform3f(P.u.CF, ...CF); gl.uniform3f(P.u.SUN, ...SUN); gl.uniform1f(P.u.FOC, 2.1); gl.uniform1f(P.u.A, W / H); };
    gl.bindFramebuffer(gl.FRAMEBUFFER, null); gl.viewport(0, 0, W, H); gl.disable(gl.BLEND);
    gl.useProgram(planetP.p); set(planetP); gl.uniform1f(planetP.u.T, t); gl.uniform1f(planetP.u.RING, Math.min(1, free / N) * fade);
    gl.bindVertexArray(quad); gl.drawArrays(gl.TRIANGLES, 0, 3);
    gl.enable(gl.BLEND); gl.blendFunc(gl.ONE, gl.ONE);
    gl.useProgram(rockP.p); set(rockP); gl.uniform1f(rockP.u.PS, Math.max(1.4, H / 380)); gl.uniform1f(rockP.u.K, 0.5 * PARAMS.glow * Math.sqrt(20000 / N));
    gl.bindVertexArray(rv);
    gl.bindBuffer(gl.ARRAY_BUFFER, pb); gl.bufferSubData(gl.ARRAY_BUFFER, 0, pos);
    gl.bindBuffer(gl.ARRAY_BUFFER, lb); gl.bufferSubData(gl.ARRAY_BUFFER, 0, lit);
    gl.drawArrays(gl.POINTS, 0, N); gl.bindVertexArray(null); gl.disable(gl.BLEND);
    raf = requestAnimationFrame(frame);
  }
  raf = requestAnimationFrame(frame);
  return () => { cancelAnimationFrame(raf); ro.disconnect(); gl.getExtension('WEBGL_lose_context')?.loseContext(); canvas.remove(); };
}

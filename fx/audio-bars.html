Spectrum analyser

Spectrum analyser is a Canvas effect: synthesised spectrum, fast-attack bars, peak caps falling under gravity. One plain JavaScript file with no dependencies; add it with npx motion-library add audio-bars.

Open it live, tune it and copy the code

What is Spectrum analyser?

Segmented green-to-red bars dance to a beat, each topped by a peak cap that hangs, then falls.

How does Spectrum analyser work?

A synthetic spectrum (pink-noise slope plus kick, snare, hi-hats and notes) drives bars that rise fast and fall slowly; caps hold a third of a second, then drop under gravity.

How do I add Spectrum analyser to my website?

  1. In your project folder, run npx motion-library add audio-bars. It writes animations/audio-bars.js. Needs Node 18 or newer.
  2. Import it and call it with a container element that has a size: import fx from './animations/audio-bars.js'; const stop = fx(document.querySelector('#hero'));
  3. Call stop() to clean up, for example when a component unmounts.

Every setting below is also a flag, so you can install it already tuned: npx motion-library add audio-bars --bars 64.

// Spectrum analyser with peak hold, no microphone: the spectrum is synthesised. A pink-noise slope, a kick that
// thumps the low bins on every beat, a snare on two and four, hi-hats on the eighths, and a bass and a lead note
// that hop between bands. Bars rise fast and fall slowly like a real meter; each peak cap waits a moment, then drops
// under gravity until a bar pushes it back up.
const PARAMS = { bars: 32, bpm: 124, gravity: 1 };

export default function (stage) {
  const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;
  const canvas = document.createElement('canvas');
  canvas.style.cssText = 'position:absolute;inset:0;width:100%;height:100%;display:block';
  stage.append(canvas);
  const ctx = canvas.getContext('2d');
  let W = 0, H = 0, D = 1;
  const resize = () => { D = Math.min(devicePixelRatio, 2); W = stage.clientWidth || 1; H = stage.clientHeight || 1; canvas.width = W * D | 0; canvas.height = H * D | 0; };
  const ro = new ResizeObserver(resize); ro.observe(stage); resize();

  const n = Math.max(8, Math.min(64, Math.round(PARAMS.bars)));
  const level = new Float32Array(n), peak = new Float32Array(n), vel = new Float32Array(n), hold = new Float32Array(n), jit = new Float32Array(n);
  const gauss = (x, m, s) => Math.exp(-((x - m) * (x - m)) / (2 * s * s));
  const BASS = [0.1, 0.1, 0.16, 0.13], LEAD = [0.48, 0.56, 0.62, 0.52, 0.7, 0.6, 0.56, 0.44];
  function spectrum(t, out) {
    const beat = t * PARAMS.bpm / 60, fb = beat % 1, bar = Math.floor(beat);
    const kick = Math.exp(-fb * 7), snare = (bar % 2 === 1) ? Math.exp(-fb * 6) : 0;
    const e8 = (beat * 2) % 1, hat = Math.exp(-e8 * 16) * (Math.floor(beat * 2) % 2 ? 1 : 0.55);
    const bassF = BASS[Math.floor(beat / 2) % 4], leadF = LEAD[Math.floor(beat) % 8], leadE = 0.55 + 0.45 * Math.exp(-fb * 3);
    for (let i = 0; i < n; i++) {
      const f = i / (n - 1);
      jit[i] += ((Math.random() - 0.5) * 0.5 - jit[i]) * 0.25;            // per-band noise, smoothed in time
      let v = 0.5 * Math.pow(1 - f, 0.9) + 0.06;
      v += 0.75 * kick * gauss(f, 0.02, 0.07);
      v += 0.45 * gauss(f, bassF, 0.035) * (0.6 + 0.4 * kick);
      v += 0.5 * snare * (gauss(f, 0.42, 0.16) * 0.9 + 0.25 * f);
      v += 0.42 * hat * Math.max(0, (f - 0.62) / 0.38);
      v += 0.42 * leadE * gauss(f, leadF, 0.025) + 0.2 * leadE * gauss(f, leadF * 1.5, 0.02);
      out[i] = Math.max(0, Math.min(1, v * 1.22 * (0.85 + jit[i] * 0.4)));
    }
  }
  const target = new Float32Array(n);
  let raf, last = performance.now(), t = 0, fresh = true;
  function frame(now) {
    const dt = Math.min(0.05, (now - last) / 1000); last = now;
    if (!reduce || fresh) {
      t += reduce ? 2.3 : dt;
      spectrum(t, target);
      for (let i = 0; i < n; i++) {
        const tg = target[i];
        level[i] = tg > level[i] ? level[i] + (tg - level[i]) * (1 - Math.exp(-dt * 40)) : level[i] * Math.exp(-dt * 4.5);
        if (reduce) level[i] = tg;
        if (level[i] >= peak[i]) { peak[i] = level[i]; vel[i] = 0; hold[i] = 0.35; }
        else if ((hold[i] -= dt) < 0) { vel[i] += 2.2 * PARAMS.gravity * dt; peak[i] = Math.max(level[i], peak[i] - vel[i] * dt); }
      }
      fresh = false;
    }

    ctx.setTransform(D, 0, 0, D, 0, 0);
    ctx.fillStyle = '#04050a'; ctx.fillRect(0, 0, W, H);
    const left = W * 0.07, right = W * 0.93, baseY = H * 0.7, topY = H * 0.1;
    const slot = (right - left) / n, bw = Math.max(2, Math.floor(slot * 0.72)), segH = Math.max(3, Math.round((baseY - topY) / 34)), gap = Math.max(1, Math.round(segH * 0.3));
    const segs = Math.floor((baseY - topY) / segH);
    const colorAt = k => { const f = k / (segs - 1); return f < 0.6 ? `hsl(${150 - f * 80},100%,${52 + f * 6}%)` : f < 0.85 ? `hsl(${102 - (f - 0.6) / 0.25 * 62},100%,55%)` : `hsl(${40 - (f - 0.85) / 0.15 * 38},100%,56%)`; };
    // segments row by row (one colour per row keeps state changes low): lit, unlit ghost, and a dim reflection
    for (let k = 0; k < segs; k++) {
      const y = Math.round(baseY - (k + 1) * segH), c = colorAt(k), thr = (k + 0.5) / segs;
      ctx.fillStyle = c;
      for (let pass = 0; pass < 3; pass++) {
        ctx.globalAlpha = pass === 0 ? 1 : pass === 1 ? 0.07 : 0.16 * (1 - k / segs);
        ctx.beginPath();
        for (let i = 0; i < n; i++) {
          const on = level[i] >= thr;
          if (pass === 0 && !on || pass === 1 && on || pass === 2 && !on) continue;
          const x = Math.round(left + i * slot + (slot - bw) / 2);
          if (pass === 2) ctx.rect(x, Math.round(baseY + gap * 2 + k * segH * 0.4), bw, Math.max(1, (segH - gap) * 0.4));   // squashed reflection
          else ctx.rect(x, y, bw, segH - gap);
        }
        ctx.fill();
      }
    }
    // peak caps with a soft glow
    ctx.globalAlpha = 1; ctx.shadowBlur = 10 * D;
    for (let i = 0; i < n; i++) {
      const k = Math.min(segs - 1, Math.max(0, Math.round(peak[i] * segs) - 0)), x = Math.round(left + i * slot + (slot - bw) / 2);
      const y = Math.round(baseY - (k + 1) * segH - segH * 0.15);
      if (peak[i] < 0.02) continue;
      ctx.fillStyle = '#f4fbff'; ctx.shadowColor = colorAt(k);
      ctx.fillRect(x, y, bw, Math.max(2, Math.round((segH - gap) * 0.55)));
    }
    ctx.shadowBlur = 0;
    // baseline
    ctx.fillStyle = 'rgba(255,255,255,0.12)'; ctx.fillRect(left, Math.round(baseY + gap / 2), right - left, 1);
    raf = requestAnimationFrame(frame);
  }
  raf = requestAnimationFrame(frame);
  return () => { cancelAnimationFrame(raf); ro.disconnect(); canvas.remove(); };
}

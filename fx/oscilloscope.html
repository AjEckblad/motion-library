Oscilloscope

Oscilloscope is a Canvas effect: x-Y Lissajous beam with speed-dependent brightness and phosphor afterglow. One plain JavaScript file with no dependencies; add it with npx motion-library add oscilloscope.

Open it live, tune it and copy the code

What is Oscilloscope?

A green CRT trace draws drifting Lissajous figures over a graticule, leaving a fading phosphor afterglow. Left–right picks the frequency ratio, up–down the phase.

How does Oscilloscope work?

Two sine generators drive x and y; each segment’s brightness is inverse to its length, as with a real beam, and each frame only partly fades the last.

How do I add Oscilloscope to my website?

  1. In your project folder, run npx motion-library add oscilloscope. It writes animations/oscilloscope.js. Needs Node 18 or newer.
  2. Import it and call it with a container element that has a size: import fx from './animations/oscilloscope.js'; const stop = fx(document.querySelector('#hero'));
  3. Call stop() to clean up, for example when a component unmounts.

Every setting below is also a flag, so you can install it already tuned: npx motion-library add oscilloscope --persistence 0.97.

// Oscilloscope in X-Y mode: two sine generators drive the beam’s x and y, tracing a Lissajous figure. Like a real
// CRT, brightness is inversely proportional to beam speed (slow turns glow hot, fast stretches go faint), and the
// phosphor keeps glowing: each frame only partly fades the last, so the drifting phase leaves an afterglow. The
// graticule sits on its own layer, so it never smears. Every few seconds the frequency ratio changes.
// Point at it to take the knobs: across picks the frequency ratio (snapping to small whole numbers), up and down the phase.
const PARAMS = { persistence: 0.85, drift: 1, glow: 1 };

export default function (stage) {
  const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;
  const mk = () => { const c = document.createElement('canvas'); c.style.cssText = 'position:absolute;inset:0;width:100%;height:100%;display:block'; stage.append(c); return c; };
  const canvas = mk(), over = mk();
  over.style.mixBlendMode = 'screen';                                  // tint + graticule added on top of the black phosphor layer
  const vig = document.createElement('div');
  vig.style.cssText = 'position:absolute;inset:0;pointer-events:none;background:radial-gradient(ellipse at 50% 46%,transparent 45%,rgba(0,0,0,.55) 100%)';
  stage.append(vig);
  const ctx = canvas.getContext('2d', { alpha: false }), octx = over.getContext('2d');
  const BG = '#020805';
  let W = 0, H = 0, D = 1, cx = 0, cy = 0, div = 0;
  const RATIOS = [[3, 2], [5, 4], [1, 3], [4, 3], [2, 1], [5, 3]];
  const KNOB = [[1, 2], [2, 3], [3, 4], [1, 1], [5, 4], [4, 3], [3, 2], [5, 3], [2, 1], [3, 1]];   // left to right, by x : y
  let ri = 0, cur = RATIOS[0], ki = -1, live = false, shownPh = -1;

  function drawGraticule() {
    octx.setTransform(D, 0, 0, D, 0, 0); octx.fillStyle = BG; octx.fillRect(0, 0, W, H);
    const gw = div * 10, gh = div * 8, x0 = Math.round(cx - gw / 2) + 0.5, y0 = Math.round(cy - gh / 2) + 0.5;
    octx.strokeStyle = 'rgba(140,220,180,0.16)'; octx.lineWidth = 1; octx.beginPath();
    for (let i = 0; i <= 10; i++) { const x = Math.round(x0 + i * div) + 0.5 - 0.5; octx.moveTo(x, y0); octx.lineTo(x, y0 + gh); }
    for (let j = 0; j <= 8; j++) { const y = Math.round(y0 + j * div) + 0.5 - 0.5; octx.moveTo(x0, y); octx.lineTo(x0 + gw, y); }
    octx.stroke();
    // minor ticks on the centre axes, five per division
    octx.strokeStyle = 'rgba(140,220,180,0.3)'; octx.beginPath();
    for (let i = 0; i <= 50; i++) { const x = Math.round(x0 + i * div / 5); octx.moveTo(x, cy - 4); octx.lineTo(x, cy + 4); }
    for (let j = 0; j <= 40; j++) { const y = Math.round(y0 + j * div / 5); octx.moveTo(cx - 4, y); octx.lineTo(cx + 4, y); }
    octx.stroke();
    octx.strokeStyle = 'rgba(140,220,180,0.26)'; octx.beginPath();
    octx.moveTo(x0, Math.round(cy) + 0.5); octx.lineTo(x0 + gw, Math.round(cy) + 0.5);
    octx.moveTo(Math.round(cx) + 0.5, y0); octx.lineTo(Math.round(cx) + 0.5, y0 + gh); octx.stroke();
    // glass vignette
    const g = octx.createRadialGradient(cx, cy, Math.min(W, H) * 0.35, cx, cy, Math.max(W, H) * 0.75);
    g.addColorStop(0, 'rgba(0,0,0,0)'); g.addColorStop(1, 'rgba(0,0,0,0.55)');
    octx.fillStyle = g; octx.fillRect(0, 0, W, H);
  }
  function drawLabels() {
    const fs = Math.max(10, Math.round(div * 0.17));
    octx.save(); octx.setTransform(D, 0, 0, D, 0, 0);
    octx.fillStyle = BG; octx.fillRect(0, H - fs * 2.6, W, fs * 2.6);
    octx.font = `${fs}px ui-monospace, SFMono-Regular, Menlo, monospace`; octx.fillStyle = 'rgba(150,240,190,0.55)'; octx.textBaseline = 'middle';
    const [a, b] = cur, y = H - fs * 1.4, x0 = cx - div * 5;
    octx.fillText('CH1 0.5V/div', x0, y); octx.fillText('CH2 0.5V/div', x0 + div * 2.6, y);
    octx.textAlign = 'right'; octx.fillText(live ? `φ ${String(shownPh).padStart(3, '0')}°   X-Y   ${a}:${b}` : `X-Y   ${a}:${b}`, x0 + div * 10, y);
    octx.restore();
  }
  const resize = () => {
    D = Math.min(devicePixelRatio, 2); W = stage.clientWidth || 1; H = stage.clientHeight || 1;
    for (const c of [canvas, over]) { c.width = W * D | 0; c.height = H * D | 0; }
    ctx.setTransform(D, 0, 0, D, 0, 0); ctx.fillStyle = '#000'; ctx.fillRect(0, 0, W, H);
    div = Math.floor(Math.min(W / 10.8, (H - 40) / 8.6)); cx = W / 2; cy = (H - 20) / 2;
    drawGraticule(); drawLabels();
  };
  const ro = new ResizeObserver(resize); ro.observe(stage); resize();

  const N = 1400, xs = new Float32Array(N + 1), ys = new Float32Array(N + 1);
  // the trace in unit coordinates as last drawn (ux, uy), and the frozen trace a ratio change morphs away from (fx, fy)
  const ux = new Float32Array(N + 1), uy = new Float32Array(N + 1), fx = new Float32Array(N + 1), fy = new Float32Array(N + 1);
  for (let i = 0; i <= N; i++) { const s = i / N * Math.PI * 2; ux[i] = fx[i] = Math.sin(cur[0] * s); uy[i] = fy[i] = Math.sin(cur[1] * s); }
  const BUCKETS = 6, paths = Array.from({ length: BUCKETS }, () => []);
  let raf, last = performance.now(), t = 0, phase = 0, switchAt = 7, mixK = 1, mixRate = 0.8, lastMove = -1e9, px = 0.5, py = 0.5;
  const retarget = (r, rate) => { if (r === cur) return; fx.set(ux); fy.set(uy); cur = r; mixK = 0; mixRate = rate; drawLabels(); };
  const onMove = e => { const b = stage.getBoundingClientRect(); px = (e.clientX - b.left) / b.width; py = (e.clientY - b.top) / b.height; lastMove = performance.now(); };
  const onLeave = () => { lastMove = -1e9; };
  stage.addEventListener('pointermove', onMove); stage.addEventListener('pointerdown', onMove); stage.addEventListener('pointerleave', onLeave);
  function frame(now) {
    const dt = Math.min(0.05, (now - last) / 1000); last = now;
    const sp = reduce ? 0.06 : 1; t += dt * sp;
    const wasLive = live; live = now - lastMove < 2500;
    if (live) {
      // across: the ratio knob, snapping to the nearest slot but only letting go a quarter-slot past its edge
      const f = Math.min(0.999, Math.max(0, px)) * KNOB.length;
      if (ki < 0 || f < ki - 0.25 || f > ki + 1.25) ki = f | 0;
      retarget(KNOB[ki], 3);
      // up and down: the phase, top = 180°, bottom = 0°, eased the short way round
      const want = (1 - Math.min(1, Math.max(0, py))) * Math.PI;
      let dp = (want - phase) % (Math.PI * 2); if (dp > Math.PI) dp -= Math.PI * 2; if (dp < -Math.PI) dp += Math.PI * 2;
      phase += dp * (1 - Math.exp(-dt * 10));
      const deg = Math.round(((phase % (Math.PI * 2)) + Math.PI * 2) % (Math.PI * 2) * 180 / Math.PI / 5) * 5 % 360;
      if (deg !== shownPh || !wasLive) { shownPh = deg; drawLabels(); }
    } else {
      if (wasLive) { ki = -1; switchAt = t + 4; drawLabels(); }
      phase += dt * sp * 0.55 * PARAMS.drift;
      if (!reduce && t > switchAt) { ri = (ri + 1) % RATIOS.length; switchAt = t + 7; retarget(RATIOS[ri], 0.8); }
    }
    mixK = Math.min(1, mixK + dt * mixRate);
    const m = mixK * mixK * (3 - 2 * mixK), a2 = cur[0], b2 = cur[1];
    const A = div * 3.3;
    for (let i = 0; i <= N; i++) {
      const s = i / N * Math.PI * 2;
      const x = fx[i] + (Math.sin(a2 * s + phase) - fx[i]) * m, y = fy[i] + (Math.sin(b2 * s) - fy[i]) * m;
      const jit = reduce ? 0 : (Math.random() - 0.5) * 0.004;
      ux[i] = x; uy[i] = y;
      xs[i] = cx + A * (x + jit); ys[i] = cy - A * (y + jit);
    }
    // fade the phosphor, then add this frame's beam
    ctx.globalCompositeOperation = 'source-over';
    ctx.fillStyle = `rgba(0,0,0,${1 - Math.min(0.97, PARAMS.persistence)})`; ctx.fillRect(0, 0, W, H);
    // an 8-bit alpha fade stalls a few levels above black; subtracting one level a frame (difference with 1) clears it
    ctx.globalCompositeOperation = 'difference'; ctx.fillStyle = 'rgb(1,1,1)'; ctx.fillRect(0, 0, W, H);
    for (const p of paths) p.length = 0;
    const ref = A * Math.PI * 2 / N;                                   // segment length of an average-speed beam
    for (let i = 0; i < N; i++) {
      const len = Math.hypot(xs[i + 1] - xs[i], ys[i + 1] - ys[i]);
      const I = Math.min(1, 0.55 * ref / (len + 0.02 * ref));             // slow beam = bright
      paths[Math.min(BUCKETS - 1, I * BUCKETS | 0)].push(i);
    }
    ctx.globalCompositeOperation = 'lighter'; ctx.lineCap = 'round';
    for (let b = 0; b < BUCKETS; b++) {
      const p = paths[b]; if (!p.length) continue;
      const I = (b + 0.7) / BUCKETS;
      ctx.beginPath();
      for (const i of p) { ctx.moveTo(xs[i], ys[i]); ctx.lineTo(xs[i + 1], ys[i + 1]); }
      ctx.strokeStyle = `rgba(60,255,150,${0.05 * I * PARAMS.glow})`; ctx.lineWidth = 9; ctx.stroke();
      ctx.strokeStyle = `rgba(80,255,160,${0.32 * I})`; ctx.lineWidth = 3; ctx.stroke();
      ctx.strokeStyle = `rgba(210,255,225,${0.85 * I * I})`; ctx.lineWidth = 1.1; ctx.stroke();
    }
    ctx.globalCompositeOperation = 'source-over';
    raf = requestAnimationFrame(frame);
  }
  raf = requestAnimationFrame(frame);
  return () => { cancelAnimationFrame(raf); ro.disconnect(); stage.removeEventListener('pointermove', onMove); stage.removeEventListener('pointerdown', onMove); stage.removeEventListener('pointerleave', onLeave); canvas.remove(); over.remove(); vig.remove(); };
}

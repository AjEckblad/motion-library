Open ocean

Open ocean is a WebGL effect: raise the wind from a sparkling calm to a grey gale. One plain JavaScript file with no dependencies; add it with npx motion-library add ocean.

Open it live, tune it and copy the code

What is Open ocean?

The open sea from a small boat. In light air the sky is clear, the water deep blue and the sun breaks into thousands of sparkles; as the wind rises, clouds gather and whitecaps break, and at the top tall slate waves hide the horizon under a grey sky.

How does Open ocean work?

Four GPU FFTs turn a wave spectrum for the wind into moving waves, and foam forms where the surface folds. Each patch of water draws how many tiny facets catch the sun, so the glitter comes as single points. Wind sets the sea and the weather, chop sharpens the crests, sun sets its height; move the pointer to slide it.

How do I add Open ocean to my website?

  1. In your project folder, run npx motion-library add ocean. It writes animations/ocean.js. Needs Node 18 or newer.
  2. Import it and call it with a container element that has a size: import fx from './animations/ocean.js'; const stop = fx(document.querySelector('#hero'));
  3. Call stop() to clean up, for example when a component unmounts.

Every setting below is also a flag, so you can install it already tuned: npx motion-library add ocean --wind 20.

// Open ocean: the open sea seen from a small boat, and the weather follows the wind. In light air the sky is clear and
// pale blue, the water a deep ultramarine with small short-crested waves running every way, and the sun (up and off to
// the left) breaks into thousands of separate sparkles on the facets that face it, thickest toward the horizon. As
// the wind rises, clouds gather, the waves grow and line up with the wind, whitecaps break; at the top end it is a
// gale under a flat grey overcast: steep dark slate waves taller than the boat that hide the horizon, thin crests
// glowing bottle green where light passes through them, foam tumbling down their faces and streaking downwind. The
// boat rides the long waves. Move the pointer and the sun (or, under cloud, its bright patch) follows it.
// The trick: the surface is an FFT ocean (Tessendorf). A JONSWAP spectrum for the chosen wind and fetch (plus a swell
// once the wind is up), spread round the wind more or less widely, gives a random amplitude to every wave vector; each
// frame those 262,000 waves are advanced in time and turned into heights and sideways (Gerstner) shifts by inverse FFTs
// on the GPU (four tiles of different sizes side by side in one half-float texture, so all four share the same 16
// passes); slopes and stretch come from neighbouring texels. Each tile is read where its own waves moved the water from,
// so the long waves never smear the short ones into streaks. Where the sideways shifts fold the surface (the Jacobian
// dips) foam is injected and then fades. Slopes too small to see in a pixel are kept as roughness (LEAN mapping). Sun glitter is counted, not smeared: each
// patch of water about a pixel wide holds a known number of tiny facets, and a Poisson draw (anchored to the moving
// water, re-rolled a few times a second) decides how many of them happen to point the sun at the eye, so near water
// shows rare bright points and far water a dense shimmer. Foam has a thickness: thick where fresh, thinning into a veil
// with round holes, combed into long lines in a gale. The boat's motion is the long waves summed at one point.
const PARAMS = { wind: 10, chop: 1.15, sun: 20 };

export default function (stage) {
  const canvas = document.createElement('canvas');
  canvas.style.cssText = 'position:absolute;inset:0;width:100%;height:100%;display:block;touch-action:pan-y';
  stage.append(canvas);
  const gl = canvas.getContext('webgl2', { antialias: false, depth: false, alpha: false, powerPreference: 'high-performance' });
  if (!gl) { stage.textContent = 'This needs WebGL2.'; return () => canvas.remove(); }
  if (!gl.getExtension('EXT_color_buffer_float')) { stage.textContent = 'This needs WebGL2 float render targets.'; gl.getExtension('WEBGL_lose_context')?.loseContext(); return () => canvas.remove(); }
  const aniso = gl.getExtension('EXT_texture_filter_anisotropic');
  const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;

  const N = 256, G = 9.81, TREP = 200;              // FFT size; the whole sea repeats exactly every 200 s
  const CASC = [{ L: 509, lo: 0, hi: 0.44 }, { L: 87, lo: 0.44, hi: 2.9 }, { L: 13.3, lo: 2.9, hi: 12.5 }, { L: 3.07, lo: 12.5, hi: 1e9 }];
  const NC = CASC.length;

  // ---------------------------------------------------------------- the weather, from the wind
  const clamp = (x, a, b) => Math.max(a, Math.min(b, x)), mix = (a, b, t) => a + (b - a) * t;
  const ss = (a, b, x) => { const t = clamp((x - a) / (b - a), 0, 1); return t * t * (3 - 2 * t); };
  const U10 = clamp(PARAMS.wind, 2, 20);
  const WX = clamp((U10 - 3) / 15, 0, 1);            // 0 light air .. 1 gale
  const CV = ss(0.18, 0.85, WX);                     // cloud cover: none, fair-weather cumulus, a closed grey deck
  const SV = 1 - ss(0.32, 0.8, WX);                  // how much of the sun gets through
  const FOAM = ss(0.14, 0.42, WX);                   // whitecaps start in a moderate breeze

  // ---------------------------------------------------------------- GLSL
  const HEAD = `#version 300 es
  precision highp float; precision highp int; precision highp sampler2D;
  vec2 cm(vec2 a, vec2 b) { return vec2(a.x * b.x - a.y * b.y, a.x * b.y + a.y * b.x); }
  vec2 ci(vec2 a) { return vec2(-a.y, a.x); }
  `;
  const VQ = `#version 300 es
  in vec2 p; uniform float Z; void main() { gl_Position = vec4(p, Z, 1); }`;

  // every wave vector of all four tiles advanced in time: h(k,t) = h0(k) e^{iwt} + conj(h0(-k)) e^{-iwt}, then the
  // spectra of the sideways shifts (packed as one complex number) and of the height
  const FS_SPEC = HEAD + `
  uniform sampler2D H0; uniform vec4 LS; uniform float TT;
  out vec4 o0;
  void main() {
    ivec2 p = ivec2(gl_FragCoord.xy); const int n = ${N};
    int c = p.x / n; ivec2 lp = ivec2(p.x - c * n, p.y);
    float L = c == 0 ? LS.x : c == 1 ? LS.y : c == 2 ? LS.z : LS.w;
    vec2 k = 6.2831853 * vec2(lp.x < n / 2 ? lp.x : lp.x - n, lp.y < n / 2 ? lp.y : lp.y - n) / L;
    float kl = length(k);
    if (kl < 1e-6) { o0 = vec4(0); return; }
    float q = floor(sqrt(9.81 * kl) * ${TREP.toFixed(1)} / 6.2831853 + .5);     // w quantised: the sea loops exactly
    float ph = 6.2831853 * fract(q * TT / ${TREP.toFixed(1)});
    vec4 h = texelFetch(H0, p, 0); vec2 e = vec2(cos(ph), sin(ph));
    vec2 ht = cm(h.xy, e) + cm(h.zw, vec2(e.x, -e.y)), iht = ci(ht), u = k / kl;
    o0 = vec4(-u.x * iht + ci(-u.y * iht), ht);
  }`;
  // one radix-2 Stockham pass of the inverse FFT, along rows (HZ = 1, inside each tile) or columns
  const FS_FFT = HEAD + `
  uniform sampler2D A; uniform int S, HZ;
  out vec4 o0;
  void main() {
    ivec2 p = ivec2(gl_FragCoord.xy); const int n = ${N};
    int i = HZ == 1 ? p.x % n : p.y, b0 = HZ == 1 ? p.x - i : 0, hs = S / 2, e = (i / S) * hs + i % hs;
    ivec2 pe = HZ == 1 ? ivec2(b0 + e, p.y) : ivec2(p.x, e), po = pe + (HZ == 1 ? ivec2(n / 2, 0) : ivec2(0, n / 2));
    float a = 6.2831853 * float(i % S) / float(S); vec2 w = vec2(cos(a), sin(a));
    vec4 x = texelFetch(A, pe, 0), y = texelFetch(A, po, 0);
    o0 = vec4(x.xy + cm(w, y.xy), x.zw + cm(w, y.zw));
  }`;
  // unpack one tile: displacement, then slopes and stretch by central differences; foam is injected where the surface
  // folds (the Jacobian dips) and fades over seconds
  const FS_ASM = HEAD + `
  uniform sampler2D A, MP; uniform int OFF; uniform float CH, DT, FI, FT, FD, DX;
  layout(location = 0) out vec4 oD; layout(location = 1) out vec4 oG; layout(location = 2) out vec4 oM;
  vec3 at(ivec2 q) { q = (q + ${N}) % ${N}; return texelFetch(A, q + ivec2(OFF, 0), 0).xyz; }     // (Dx, Dz, height)
  void main() {
    ivec2 p = ivec2(gl_FragCoord.xy);
    vec3 c = at(p), xp = at(p + ivec2(1, 0)), xm = at(p - ivec2(1, 0)), zp = at(p + ivec2(0, 1)), zm = at(p - ivec2(0, 1));
    vec3 gx = (xp - xm) / (2. * DX), gz = (zp - zm) / (2. * DX);
    oD = vec4(c.x * CH, c.z, c.y * CH, gz.x * CH);
    oG = vec4(gx.z, gz.z, gx.x * CH, gz.y * CH);
    float J = (1. + oG.z) * (1. + oG.w) - oD.w * oD.w;
    float f = texelFetch(MP, p, 0).z;
    f = f * exp(-DT * FD) + FI * DT * 4.5 * clamp((FT - J) * 6., 0., 1.);
    oM = vec4(gx.z * gx.z, gz.z * gz.z, min(f, 1.6), J);
  }`;

  // the sky: single scattering through a real atmosphere (Rayleigh, Mie haze and ozone), stored by azimuth and the
  // square root of elevation; a second copy adds the clouds, for the water to reflect
  const ATMO = `
  const float RE = 6360e3, RA = 6420e3, PI = 3.14159265;
  const vec3 BR = vec3(5.802e-6, 13.558e-6, 33.1e-6), BO = vec3(.65e-6, 1.881e-6, .085e-6);
  uniform vec3 SD, SC, SKT; uniform float SUNI, HZM;
  float sph(vec3 o, vec3 d, float r) { float b = dot(o, d), c = dot(o, o) - r * r, h = b * b - c; return h < 0. ? -1. : -b + sqrt(h); }
  float sphN(vec3 o, vec3 d, float r) { float b = dot(o, d), c = dot(o, o) - r * r, h = b * b - c; return h < 0. ? -1. : -b - sqrt(h); }
  vec3 dens(float a) { return vec3(exp(-a / 8000.), exp(-a / 1100.), max(0., 1. - abs(a - 25000.) / 15000.)); }
  vec3 ext(vec3 d) { return BR * d.x + 5.6e-6 * HZM * d.y + BO * d.z; }
  vec3 transTop(vec3 o, vec3 d) {
    if (sphN(o, d, RE) > 0.) return vec3(0);
    float t = sph(o, d, RA); vec3 od = vec3(0);
    for (int i = 0; i < 5; i++) { float s0 = float(i) / 5., s1 = float(i + 1) / 5.; vec3 q = o + d * t * (s0 * s0 + s1 * s1) * .5; od += dens(length(q) - RE) * t * (s1 * s1 - s0 * s0); }
    return exp(-ext(od));
  }
  vec3 atmo(vec3 rd) {
    vec3 o = vec3(0, RE + 8., 0); rd.y = max(rd.y, .0); rd = normalize(rd);
    float t = sph(o, rd, RA); vec3 sR = vec3(0), sM = vec3(0), od = vec3(0);
    for (int i = 0; i < 14; i++) {
      float s0 = float(i) / 14., s1 = float(i + 1) / 14., dt = t * (s1 * s1 - s0 * s0);
      vec3 q = o + rd * t * (s0 * s0 + s1 * s1) * .5, dn = dens(length(q) - RE) * dt;
      od += dn * .5; vec3 T = exp(-ext(od)) * transTop(q, SD); od += dn * .5;
      sR += T * dn.x; sM += T * dn.y;
    }
    float mu = dot(rd, SD), g = .8;
    float pR = 3. / (16. * PI) * (1. + mu * mu), pM = (1. - g * g) / (4. * PI * pow(1. + g * g - 2. * g * mu, 1.5));
    vec3 ms = SUNI * BR * 1500. * max(SD.y + .1, 0.) * (1. - .3 * rd.y);  // light scattered more than once: blue, even
    return (SUNI * (sR * BR * pR + sM * 5e-6 * HZM * pM) + ms) * SKT;
  }
  vec2 lutUV(vec3 r) { return vec2(fract(atan(r.z, r.x) / 6.2831853), sqrt(clamp(asin(clamp(r.y, 0., 1.)) / 1.5707963, 0., 1.))); }
  vec3 lutDir(vec2 uv) { float az = uv.x * 6.2831853, el = uv.y * uv.y * 1.5707963; return vec3(cos(az) * cos(el), sin(el), sin(az) * cos(el)); }
  `;
  const CLOUDS = `
  uniform float CT, CV;
  float hsh(vec2 p) { vec3 q = fract(p.xyx * .1031); q += dot(q, q.yzx + 33.33); return fract((q.x + q.y) * q.z); }
  float vn(vec2 p) { vec2 i = floor(p), f = fract(p); f = f * f * (3. - 2. * f);
    return mix(mix(hsh(i), hsh(i + vec2(1, 0)), f.x), mix(hsh(i + vec2(0, 1)), hsh(i + 1.), f.x), f.y); }
  float fbm(vec2 p, int oc) { float s = 0., a = .5; mat2 m = mat2(1.6, 1.2, -1.2, 1.6);
    for (int i = 0; i < 7; i++) { if (i >= oc) break; s += a * vn(p); p = m * p; a *= .5; } return s; }
  // clouds a kilometre and a half up: none in light air, then fair-weather heaps, then a closed deck. The deck is an
  // even, slightly warm grey, a little darker where it is thick; the hidden sun only shows as a brighter patch
  vec4 clouds(vec3 rd, int oc) {
    if (rd.y < .003 || CV < .005) return vec4(0);
    float t = 1500. / rd.y; vec2 p = rd.xz * t;
    vec2 q = p * .00042 + vec2(CT * .0045, -CT * .0021);
    float d = fbm(q, oc), th = mix(.8, -.25, CV);
    float a = smoothstep(th, th + .2 + .25 * CV, d) * exp(-t / 150000.);
    if (a < .001) return vec4(0);
    vec2 so = SD.xz / max(length(SD.xz), 1e-3) * .11;
    float ds = fbm(q + so, max(oc - 2, 2)), lit = clamp(.5 + (d - ds) * 3., 0., 1.);
    float mu = dot(rd, SD), thick = smoothstep(th + .1, th + .7, d);
    float mot = (fbm(q * .35 + 3.1, 3) - .5) * exp(-t / 30000.);      // the deck: broad, soft mottling, evening out with distance
    vec3 deck = vec3(1., .965, .925) * .056 * SUNI * (1. + .55 * mot) * (1. - .2 * thick * exp(-t / 30000.));
    vec3 heap = (SC * SUNI * .07 * (.35 + .65 * lit) + vec3(.6, .65, .75) * SUNI * .02) * (1. - .35 * thick);
    vec3 c = mix(heap, deck, smoothstep(.45, .95, CV));
    c += SC * SUNI * .5 * pow(max(mu, 0.), 12.) * (1. - .7 * thick) * (.4 + .6 * lit);
    return vec4(c * a, a);
  }
  `;
  const FS_LUT = HEAD + ATMO + CLOUDS + `
  uniform vec2 RES; layout(location = 0) out vec4 oA; layout(location = 1) out vec4 oC;
  void main() {
    vec3 rd = lutDir(gl_FragCoord.xy / RES); vec3 a = atmo(rd);
    vec4 c = clouds(rd, 5);
    oA = vec4(a, 1); oC = vec4(a * (1. - c.a) + c.rgb, 1);
  }`;

  // the sea: a grid laid over the screen, cast onto the curved sea surface and displaced
  const VS_SEA = HEAD + `
  in vec2 g; uniform vec3 CP, CF, CR, CU; uniform float FOC, ASP, SY0, SY1, CELL, DIP, L0, L1, GH;
  uniform sampler2D D0, D1; out vec2 vX, vX1; out vec3 vP;
  const float RE = 6.371e6, NEAR = .3, FAR = 3e4;
  void main() {
    // dense on screen, coarse in a wide margin around it, so a big wave near the camera never opens a gap
    float u = g.x * 2. - 1., au = abs(u), sx = sign(u) * (au < .88 ? au / .88 * .5 : .5 + (au - .88) / .12 * .7) * ASP;
    float w = 1. - g.y, sy = w < .86 ? mix(SY1, -.5, w / .86) : mix(-.5, SY0, (w - .86) / .14);
    vec3 rd = normalize(CF * FOC + CR * sx + CU * sy);
    // steep rays are laid on a plane just a metre under the lens, so even when the boat sits low on a crest the waves
    // right below still reach the bottom of the frame; rays toward the horizon use the true height
    float e = max(-rd.y, DIP), hr = 2. * mix(GH, min(GH, 1.), smoothstep(.12, .45, e)) / RE, d = RE * (e - sqrt(max(e * e - hr, 0.)));
    vec2 x = CP.xz + normalize(rd.xz) * d;
    float foot = d * CELL / max(e, .003);
    vec3 Da = textureLod(D0, x / L0, max(0., log2(foot * ${N}. / L0))).xyz, D = Da + textureLod(D1, x / L1, max(0., log2(foot * ${N}. / L1))).xyz;
    D.xz *= smoothstep(.5, 7., d);                   // right under the boat the sideways shifts would pull the grid away
    vec3 P = vec3(x.x + D.x, D.y - d * d / (2. * RE), x.y + D.z);
    P.y = mix(P.y, min(P.y, CP.y - .6), smoothstep(9., 3., d));   // a crest right at the boat never rises past the lens
    vX = x; vX1 = x + Da.xz; vP = P;
    vec3 q = P - CP, v = vec3(dot(q, CR), dot(q, CU), dot(q, CF));
    gl_Position = vec4(v.x * FOC * 2. / ASP, v.y * FOC * 2., v.z * (FAR + NEAR) / (FAR - NEAR) - 2. * FAR * NEAR / (FAR - NEAR), v.z);
  }`;
  const FS_SEA = HEAD + ATMO + `
  in vec2 vX, vX1; in vec3 vP; out vec4 o;
  uniform sampler2D D0, D1, D2, G0, G1, G2, G3, M0, M1, M2, M3, LA, LC; uniform vec3 CP, BODY; uniform float L0, L1, L2, L3, T;
  uniform vec2 WD; uniform float SV, CV, FI, CAP, HD, HS, KG, GK, WX;
  float hsh(vec2 p) { vec3 q = fract(p.xyx * .1031); q += dot(q, q.yzx + 33.33); return fract((q.x + q.y) * q.z); }
  float vn(vec2 p) { vec2 i = floor(p), f = fract(p); f = f * f * (3. - 2. * f);
    return mix(mix(hsh(i), hsh(i + vec2(1, 0)), f.x), mix(hsh(i + vec2(0, 1)), hsh(i + 1.), f.x), f.y); }
  // sea foam is a lace: thin white walls between bubbles of every size (the second-nearest minus the nearest cell point)
  vec2 lace(vec2 p) {
    vec2 i = floor(p), f = fract(p); float d1 = 8., d2 = 8.;
    for (int y = -1; y <= 1; y++) for (int x = -1; x <= 1; x++) {
      vec2 o = vec2(x, y), r = o + vec2(hsh(i + o), hsh(i + o + 17.3)) * .9 + .05 - f; float d = dot(r, r);
      if (d < d1) { d2 = d1; d1 = d; } else if (d < d2) d2 = d;
    }
    return vec2(sqrt(d1), sqrt(d2) - sqrt(d1));
  }
  vec3 skyC(vec3 r, float lod) { r.y = abs(r.y); return textureLod(LC, lutUV(r), lod).rgb; }
  float lam(float c, float s) { float a = c / (s * sqrt(2. * max(1. - c * c, 1e-5))); return a < 1.6 ? (1. - 1.259 * a + .396 * a * a) / (3.535 * a + 2.181 * a * a) : 0.; }
  float gpdf(vec2 z, vec2 v) { return exp(-.5 * (z.x * z.x / v.x + z.y * z.y / v.y)) / (6.2831853 * sqrt(v.x * v.y)); }
  // how many of the facets in a patch point the sun at the eye: a Poisson draw with the expected count
  float pois(float l, float u) {
    if (l > 9.) return max(0., floor(l + sqrt(l) * 3.46 * (u - .5) + .5));
    float p = exp(-l), c = p, k = 0.;
    for (int i = 0; i < 20; i++) { if (u < c) break; k += 1.; p *= l / k; c += p; }
    return k;
  }
  // the glitter: patches about a pixel across (two sizes, blended), fixed to the moving water and re-rolled a few
  // times a second; the result averages to the smooth slope density but arrives as separate points of light
  float glint(vec2 x, float cs, float pdf) {
    float l = log2(max(cs, 2e-4)), l0 = floor(l), fr = l - l0, g = 0.;
    for (int i = 0; i < 2; i++) {
      float c = exp2(l0 + float(i)), ar = c * c * GK;
      vec2 id = floor(x / c);
      float ph = hsh(id + vec2(l0 * 3.17 + float(i) * 7.7, 1.3));
      float u = hsh(id * .917 + vec2(floor(T * 6. + ph * 6.) * 1.618, c * 13.1 + 4.2));
      g += (i == 0 ? 1. - fr : fr) * pois(ar * pdf, u) / ar;
    }
    return g;
  }
  void main() {
    vec2 x = vX;
    vec2 dx = dFdx(x), dy = dFdy(x);
    float fy = length(fwidth(x)), cs = min(length(dx), length(dy));      // derivatives outside any branch
    // each tile is read where its own waves moved this bit of water from, not where the bigger waves did: so the long
    // waves' push and pull does not smear the shorter ones into streaks
    vec2 xe = vP.xz, x1 = vX1;
    vec4 g0 = texture(G0, x / L0), g1 = texture(G1, x1 / L1), g2 = texture(G2, xe / L2), g3 = texture(G3, xe / L3);
    vec4 m0 = texture(M0, x / L0), m1 = texture(M1, x1 / L1), m2 = texture(M2, xe / L2), m3 = texture(M3, xe / L3);
    vec4 d0 = texture(D0, x / L0), d1 = texture(D1, x1 / L1), d2 = texture(D2, xe / L2);
    vec2 jj = 1. + g0.zw + g1.zw + g2.zw + g3.zw;
    vec2 s = (g0.xy + g1.xy + g2.xy + g3.xy) / max(jj, vec2(.3));
    vec2 vr = max(m0.xy - g0.xy * g0.xy, 0.) + max(m1.xy - g1.xy * g1.xy, 0.) + max(m2.xy - g2.xy * g2.xy, 0.) + max(m3.xy - g3.xy * g3.xy, 0.) + CAP;
    vec3 n = normalize(vec3(-s.x, 1., -s.y));
    vec3 V = CP - vP; float dist = length(V); V /= dist;
    float nv = max(dot(n, V), .02);
    // the sky in the water, blurred by the slopes too small to resolve here; on rough water many of the small facets
    // face the eye more squarely than the mean surface, so less of the sky comes back at a glancing view
    vec3 R = reflect(-V, n);
    float sg = sqrt(.5 * (vr.x + vr.y));
    float F = .02 + .98 * pow(1. - min(nv + 1.5 * sg, 1.), 5.);
    vec3 refl = skyC(R, clamp(log2(2.2 * sg * 512. / 6.2831853), 0., 7.)) * mix(vec3(1), vec3(.86, .96, 1.08), CV);
    // the sun: a Gaussian slope distribution (resolved normal plus unresolved spread) with Smith shadowing; in sunshine
    // it is counted out as glitter, under cloud it is the veiled sun's soft sheen
    vec3 Hh = normalize(SD + V);
    vec2 zh = n.xz / n.y - Hh.xz / Hh.y;
    vec2 vd = normalize(V.xz + 1e-6), ld = normalize(SD.xz + 1e-6);
    float sv = sqrt(dot(vd * vd, vr)), sl = sqrt(dot(ld * ld, vr));
    float Gs = 1. / (1. + lam(max(V.y, .01), sv) + lam(max(SD.y, .01), sl));
    float Fh = .02 + .98 * pow(1. - max(dot(V, Hh), 0.), 5.);

  };
  const resize = () => {
    const w = stage.clientWidth || 1, h = stage.clientHeight || 1, d = Math.min(devicePixelRatio, 2, Math.max(0.5, Math.sqrt(700000 / (w * h))));
    W = canvas.width = Math.max(1, (w * d) | 0); H = canvas.height = Math.max(1, (h * d) | 0);
    freeTargets();
    scene = tex(W, H, gl.RGBA16F, { filter: gl.LINEAR });
    depthRB = gl.createRenderbuffer(); gl.bindRenderbuffer(gl.RENDERBUFFER, depthRB); gl.renderbufferStorage(gl.RENDERBUFFER, gl.DEPTH_COMPONENT24, W, H);
    sceneF = fbo([scene], depthRB);
    let bw = W, bh = H;
    for (let i = 0; i < 6; i++) { bw = Math.max(1, bw >> 1); bh = Math.max(1, bh >> 1); const t = tex(bw, bh, gl.RGBA16F, { filter: gl.LINEAR }); bloom.push({ t, f: fbo([t]), w: bw, h: bh }); }
    // the sea grid: about one vertex every 3 pixels, from below the frame to past the highest the horizon can sit
    // (the boat pitches and rolls); the top of the grid is set each frame, just above the horizon
    SYMAX = FOC * Math.tan(PITCH0 + 0.12) + 0.5 * (W / H) * Math.tan(0.12) + 0.04;
    const GX = Math.ceil(W / 0.88 / 3), GY = Math.ceil((SYMAX + 0.5) * H / 0.86 / 3);
    CELL = (SYMAX + 0.5) / (GY * 0.86) / FOC;
    const v = new Float32Array((GX + 1) * (GY + 1) * 2);
    for (let j = 0, k = 0; j <= GY; j++) for (let i = 0; i <= GX; i++) { v[k++] = i / GX; v[k++] = j / GY; }
    const idx = new Uint32Array(GX * GY * 6);
    for (let j = 0, k = 0; j < GY; j++) for (let i = 0; i < GX; i++) {
      const a = j * (GX + 1) + i, b = a + 1, c = a + GX + 1, e = c + 1;
      idx[k++] = a; idx[k++] = b; idx[k++] = e; idx[k++] = a; idx[k++] = e; idx[k++] = c;
    }
    idxCount = idx.length;
    gl.bindVertexArray(seaVAO);
    gl.bindBuffer(gl.ARRAY_BUFFER, gridBuf); gl.bufferData(gl.ARRAY_BUFFER, v, gl.STATIC_DRAW);
    gl.enableVertexAttribArray(0); gl.vertexAttribPointer(0, 2, gl.FLOAT, false, 0, 0);
    gl.bindBuffer(gl.ELEMENT_ARRAY_BUFFER, idxBuf); gl.bufferData(gl.ELEMENT_ARRAY_BUFFER, idx, gl.STATIC_DRAW);
    gl.bindVertexArray(null);
  };
  const ro = new ResizeObserver(resize); ro.observe(stage); resize();

  // ---------------------------------------------------------------- pointer: the sun follows it
  let px = 0.5, py = 0.5, lastMove = -1e9, held = false;
  const onMove = e => { const b = canvas.getBoundingClientRect(); px = (e.clientX - b.left) / b.width; py = (e.clientY - b.top) / b.height; lastMove = performance.now(); held = e.pointerType === 'mouse'; };
  const onLeave = () => { held = false; };
  canvas.addEventListener('pointermove', onMove); canvas.addEventListener('pointerdown', onMove); canvas.addEventListener('pointerleave', onLeave);
  // at rest the sun stands up and off to the left in sunshine; under cloud its bright patch sits a little left of centre
  const base = { az: -Math.PI / 2 - mix(0.58, 0.22, ss(0.3, 0.9, WX)), el: clamp(PARAMS.sun, 2, 50) * Math.PI / 180 };
  const sun = { az: base.az, el: base.el, va: 0, ve: 0 };
  let lutKey = '', drawnKey = '';

  // ---------------------------------------------------------------- frame
  let raf, last = performance.now(), t = 37;
  for (let i = 0; i < 120; i++) simulate((t - 4 + i / 30) % TREP, 1 / 30, FOAM);   // four seconds of foam history
  function frame(now) {
    const dt = Math.min(1 / 30, Math.max(0, (now - last) / 1000)); last = now;
    if (!reduce) t += dt;
    placeCam(t);
    const live = held || now - lastMove < 1800, asp = Math.min(1.6, W / H);
    const tAz = live ? -Math.PI / 2 + (px - 0.5) * 0.62 * asp : base.az;
    const tEl = live ? clamp(0.012 + (0.62 - py) * 0.5, 0.006, 0.6) : base.el;
    for (let i = 0; i < 4; i++) {
      const h = dt / 4;
      sun.va += (36 * (tAz - sun.az) - 12 * sun.va) * h; sun.az += sun.va * h;
      sun.ve += (36 * (tEl - sun.el) - 12 * sun.ve) * h; sun.el += sun.ve * h;
    }
    if (reduce) { sun.az = tAz; sun.el = tEl; }
    const SD = [Math.cos(sun.az) * Math.cos(sun.el), Math.sin(sun.el), Math.sin(sun.az) * Math.cos(sun.el)];
    const tr = sunTrans(SD), SUNI = 16;

    // reduced motion: the sea is a still; it is drawn again only when the sun or the size changes
    const stillKey = SD.map(v => v.toFixed(5)).join() + W + 'x' + H;
    if (reduce && stillKey === drawnKey) { raf = requestAnimationFrame(frame); return; }
    drawnKey = stillKey;
    if (!reduce || !simulate.done) { simulate(t % TREP, reduce ? 0 : dt, FOAM); simulate.done = true; }

    // the sky lookup, redrawn when the sun has moved or the clouds have drifted a little
    const key = SD.map(v => v.toFixed(4)).join() + '|' + Math.floor(t * 4);
    if (key !== lutKey) {
      lutKey = key;
      const u = use(P.lut); gl.bindFramebuffer(gl.FRAMEBUFFER, lutF); gl.viewport(0, 0, LW, LH);
      gl.uniform2f(u.RES, LW, LH); gl.uniform3fv(u.SD, SD); gl.uniform3fv(u.SC, tr); gl.uniform1f(u.SUNI, SUNI); gl.uniform1f(u.HZM, HZM); gl.uniform3fv(u.SKT, SKT);
      gl.uniform1f(u.CT, t); gl.uniform1f(u.CV, CV); gl.uniform1f(u.Z, 0); quad();
      gl.bindFramebuffer(gl.FRAMEBUFFER, null);
      for (const tx of [lutA, lutC]) { gl.bindTexture(gl.TEXTURE_2D, tx); gl.generateMipmap(gl.TEXTURE_2D); }
    }

    // the scene, in linear light
    gl.bindFramebuffer(gl.FRAMEBUFFER, sceneF); gl.viewport(0, 0, W, H);
    gl.clearColor(0, 0, 0, 1); gl.clearDepth(1); gl.clear(gl.COLOR_BUFFER_BIT | gl.DEPTH_BUFFER_BIT);
    gl.enable(gl.DEPTH_TEST); gl.depthFunc(gl.LESS); gl.depthMask(true);
    const SY1 = Math.min(SYMAX, FOC * Math.tan(cam.pitch - cam.dip) + 0.5 * (W / H) * Math.abs(Math.tan(cam.roll)) + 0.03);
    let u = use(P.sea);
    gl.uniform3fv(u.CP, cam.p); gl.uniform3fv(u.CF, cam.f); gl.uniform3fv(u.CR, cam.r); gl.uniform3fv(u.CU, cam.u);
    gl.uniform1f(u.FOC, FOC); gl.uniform1f(u.ASP, W / H); gl.uniform1f(u.SY0, SY0); gl.uniform1f(u.SY1, SY1); gl.uniform1f(u.CELL, CELL);
    gl.uniform1f(u.DIP, cam.dip); gl.uniform1f(u.GH, Math.max(cam.p[1], 0.5));
    gl.uniform1f(u.L0, CASC[0].L); gl.uniform1f(u.L1, CASC[1].L); gl.uniform1f(u.L2, CASC[2].L); gl.uniform1f(u.L3, CASC[3].L); gl.uniform1f(u.T, t);
    gl.uniform3fv(u.SD, SD); gl.uniform3fv(u.SC, tr); gl.uniform1f(u.SUNI, SUNI); gl.uniform1f(u.HZM, HZM); gl.uniform3fv(u.SKT, SKT); gl.uniform2f(u.WD, Math.cos(windDir), Math.sin(windDir));
    gl.uniform1f(u.SV, SV); gl.uniform1f(u.CV, CV); gl.uniform1f(u.FI, FOAM); gl.uniform1f(u.WX, WX); gl.uniform1f(u.HS, Math.max(HS, 0.2));
    gl.uniform1f(u.CAP, 0.004 + 0.0022 * U10); gl.uniform1f(u.HD, mix(26000, 2600, WX * WX)); gl.uniform1f(u.KG, 0.22); gl.uniform1f(u.GK, 12);
    const bc = [[0.004, 0.024, 0.062], [0.004, 0.022, 0.06], [0.004, 0.024, 0.056]], bi = WX < 0.5 ? 0 : 1, bt = WX < 0.5 ? WX * 2 : WX * 2 - 1;
    gl.uniform3fv(u.BODY, bc[bi].map((v, i) => mix(v, bc[bi + 1][i], bt)));
    bindTex(u.D0, 0, casc[0].d); bindTex(u.D1, 1, casc[1].d); bindTex(u.D2, 10, casc[2].d);
    bindTex(u.G0, 2, casc[0].g); bindTex(u.G1, 3, casc[1].g); bindTex(u.G2, 4, casc[2].g); bindTex(u.G3, 11, casc[3].g); bindTex(u.M3, 12, casc[3].m[casc[3].w]);
    bindTex(u.M0, 5, casc[0].m[casc[0].w]); bindTex(u.M1, 6, casc[1].m[casc[1].w]); bindTex(u.M2, 7, casc[2].m[casc[2].w]);
    bindTex(u.LA, 8, lutA); bindTex(u.LC, 9, lutC);
    gl.bindVertexArray(seaVAO); gl.drawElements(gl.TRIANGLES, idxCount, gl.UNSIGNED_INT, 0);
    gl.depthFunc(gl.LEQUAL); gl.depthMask(false);
    u = use(P.back);
    gl.uniform3fv(u.CF, cam.f); gl.uniform3fv(u.CR, cam.r); gl.uniform3fv(u.CU, cam.u); gl.uniform1f(u.FOC, FOC); gl.uniform2f(u.R, W, H);
    gl.uniform3fv(u.SD, SD); gl.uniform3fv(u.SC, tr); gl.uniform1f(u.SUNI, SUNI); gl.uniform1f(u.HZM, HZM); gl.uniform3fv(u.SKT, SKT); gl.uniform1f(u.CT, t); gl.uniform1f(u.CV, CV); gl.uniform1f(u.SV, SV); gl.uniform1f(u.Z, 0.999999);
    bindTex(u.LA, 0, lutA); quad();
    gl.disable(gl.DEPTH_TEST); gl.depthMask(true);

    // bloom down and up, then tone map to the screen
    u = use(P.down); gl.uniform1f(u.Z, 0);
    let src = scene, sw = W, sh = H;
    bloom.forEach((b, i) => {
      gl.bindFramebuffer(gl.FRAMEBUFFER, b.f); gl.viewport(0, 0, b.w, b.h);
      bindTex(u.S, 0, src); gl.uniform2f(u.TX, 1 / sw, 1 / sh); gl.uniform1f(u.TH, i === 0 ? 1.2 : 0); quad();
      src = b.t; sw = b.w; sh = b.h;
    });
    u = use(P.up); gl.uniform1f(u.Z, 0);
    gl.enable(gl.BLEND); gl.blendFunc(gl.ONE, gl.ONE);
    for (let i = bloom.length - 1; i > 0; i--) {
      const s = bloom[i], d = bloom[i - 1];
      gl.bindFramebuffer(gl.FRAMEBUFFER, d.f); gl.viewport(0, 0, d.w, d.h);
      bindTex(u.S, 0, s.t); gl.uniform2f(u.TX, 1 / s.w, 1 / s.h); quad();
    }
    gl.disable(gl.BLEND);
    u = use(P.comp); gl.bindFramebuffer(gl.FRAMEBUFFER, null); gl.viewport(0, 0, W, H);
    bindTex(u.S, 0, scene); bindTex(u.B, 1, bloom[0].t); gl.uniform2f(u.R, W, H); gl.uniform1f(u.EX, mix(0.5, 0.56, WX)); gl.uniform1f(u.DS, 0.1 * CV); gl.uniform1f(u.Z, 0); quad();
    raf = requestAnimationFrame(frame);
  }
  raf = requestAnimationFrame(frame);
  return () => {
    cancelAnimationFrame(raf); ro.disconnect();
    canvas.removeEventListener('pointermove', onMove); canvas.removeEventListener('pointerdown', onMove); canvas.removeEventListener('pointerleave', onLeave);
    freeTargets(); textures.forEach(t => gl.deleteTexture(t)); fbos.forEach(f => gl.deleteFramebuffer(f)); programs.forEach(p => gl.deleteProgram(p));
    [quadBuf, gridBuf, idxBuf].forEach(b => gl.deleteBuffer(b)); [quadVAO, seaVAO].forEach(v => gl.deleteVertexArray(v));
    gl.getExtension('WEBGL_lose_context')?.loseContext(); canvas.remove();
  };
}
